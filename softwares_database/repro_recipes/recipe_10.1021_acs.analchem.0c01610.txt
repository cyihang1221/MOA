workflow_steps: 9
  Step 1: Sample preparation and freeze-thaw cycling
    tool: manual protocol
    parameters: n_cycles=5, freezing_temperature_c=-80, thawing_condition=room temperature, cycle_duration=not specified
    input: human serum samples (n=99)
    output: serum samples subjected to 1–5 freeze-thaw cycles
    confidence: medium
  Step 2: Chemical isotope labeling (CIL)
    tool: dansylation derivatization
    parameters: labeling_reagents=['12C-dansyl chloride', '13C-dansyl chloride'], labeling_ratio=1:1 sample pairing, reaction_conditions=60°C, 30 min, pH 9–10
    input: serum samples post-FTC
    output: 12C-/13C-labeled metabolite derivatives
    confidence: high
  Step 3: LC-MS analysis
    tool: LC-MS platform
    parameters: instrument=Agilent 1290 Infinity II LC / 6545 Q-TOF MS, column=Agilent Poroshell 120 EC-C18 (2.7 μm, 2.1 × 100 mm), gradient=not fully specified, flow_rate_ml_min=0.3, column_temperature_c=40, injection_volume_ul=2, ionization_mode=ESI positive, mass_range_mz=[50, 1000], acquisition_mode=full-scan MS
    input: 12C-/13C-labeled derivatives
    output: raw .d files (Agilent format)
    confidence: medium
  Step 4: Data processing and peak pair detection
    tool: in-house software
    algorithm: isotope-labeled peak pair detection
    parameters: mass_tolerance_ppm=5, retention_time_tolerance_min=0.2, intensity_threshold=not specified, pairing_rule=12C/13C peak pairs with same RT and ±1 Da mass difference
    input: raw .d files
    output: quantified peak pair table (2790 features)
    confidence: low
  Step 5: Metabolite identification
    tool: database search + MS/MS validation
    algorithm: mass matching + spectral library matching
    parameters: database=['HMDB', 'Metlin'], mass_tolerance_ppm=5, MS_MS_similarity_threshold=not specified, confidence_levels=['positive identification', 'high-confident putative', 'mass-matched']
    input: peak pair table + MS/MS spectra
    output: annotated metabolite table (150 confirmed, 235 high-confident putative, 1949 mass-matched)
    confidence: medium
  Step 6: Relative quantification
    tool: in-house script
    algorithm: 13C/12C peak area ratio
    parameters: normalization_method=internal standard not specified, ratio_calculation=13C peak area / 12C peak area
    input: peak pair table
    output: relative abundance matrix (495 comparisons × 2790 features)
    confidence: low
  Step 7: Multivariate statistical analysis
    tool: SIMCA-P+
    algorithm: PCA, PLS-DA
    parameters: n_components=2, scaling=UV, cross_validation=7-fold, model_validation=permutation test (200 permutations)
    input: relative abundance matrix
    output: scores plots, model metrics (R2, Q2), variable importance (VIP)
    confidence: high
  Step 8: Univariate statistical testing
    tool: in-house R script or SIMCA-P+
    algorithm: t-test / ANOVA
    parameters: test_type=paired or unpaired t-test depending on comparison, p_value_threshold=0.05, correction=FDR (Benjamini-Hochberg)
    input: relative abundance matrix
    output: list of significantly different metabolites
    confidence: medium
  Step 9: FTC effect detection method
    tool: reanalysis protocol
    algorithm: additional FTC + re-measurement
    parameters: additional_cycle=1, reference_comparison=original vs. +1 FTC
    input: original serum samples
    output: FTC-effect diagnostic metric (e.g., % changed features, PCA drift)
    confidence: high
