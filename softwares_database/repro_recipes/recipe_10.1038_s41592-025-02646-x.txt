workflow_steps: 4
  Step 1: Electronic noise removal
    tool: Spectral Denoising
    algorithm: Ion count thresholding
    parameters: ion_count_threshold=4, intensity_tolerance_percent=0.1
    input: Orbitrap MS/MS spectra (raw .raw or .mzML)
    output: Electronically denoised MS/MS spectra
    confidence: high
  Step 2: Chemical noise removal
    tool: Spectral Denoising
    algorithm: Entropy-based spectral filtering
    input: Electronically denoised MS/MS spectra
    output: Chemically denoised MS/MS spectra
    confidence: medium
  Step 3: Spectral similarity scoring
    tool: Denoising Search
    algorithm: Entropy similarity
    parameters: reference_spectra_source=NIST23, similarity_metric=spectral_entropy
    input: Chemically denoised MS/MS spectra
    output: Annotated compound list with similarity scores
    confidence: high
  Step 4: False discovery rate (FDR) estimation
    tool: Denoising Search
    algorithm: Target-decoy based FDR
    parameters: top_hits_considered=3, fdr_threshold=0.01
    input: Annotated compound list with similarity scores
    output: FDR-filtered metabolite and exposome annotations
    confidence: medium
