overall_reproducibility_score: 87/100

workflow_steps: 2
  Step 1: Conversion of MS signal intensities to QC-injection-amount equivalents
    tool: PowerU
    algorithm: QC-based signal calibration using serial dilution reference
    parameters: dilution_series_used=True, QC_sample_injection_amount_reference=used as proxy for true concentration
    input: feature intensity table (from LC-MS/MS)
    output: QC-injection-amount-equivalent feature table
    confidence: medium
  Step 2: Normality assessment and adaptive statistical testing
    tool: PowerU
    algorithm: Shapiro-Wilk normality test followed by choice of t-test (if normal) or Mann-Whitney U test (if non-normal)
    parameters: alpha_normality=0.05, test_selection_rule=Shapiro-Wilk p > 0.05 → t-test; else → Mann-Whitney U
    input: QC-injection-amount-equivalent feature table
    output: list of significantly altered metabolites with test-specific p-values
    confidence: high
