paper_title: Automated biomarker candidate discovery in imaging mass spectrometry data through spatially localized Shapley additive explanations.

overall_reproducibility_score: 78/100

workflow_steps: 5
  Step 1: IMS data preprocessing
    tool: Custom Python pipeline
    algorithm: peak picking, normalization, spatial registration
    parameters: peak_picking_method=local_maxima, normalization_method=total ion current, spatial_resolution=pixel-wise
    input: raw .imzML files
    output: preprocessed feature matrix (m/z × pixel)
    confidence: low
  Step 2: Supervised classification modeling
    tool: scikit-learn
    algorithm: Random Forest classifier
    parameters: n_estimators=100, max_depth=10, random_state=42, class_weight=balanced
    input: preprocessed feature matrix
    output: trained classification model
    confidence: high
  Step 3: SHAP value computation
    tool: SHAP (TreeExplainer)
    algorithm: TreeExplainer for Random Forest
    parameters: n_jobs=-1, approximate=False
    input: trained Random Forest model + feature matrix
    output: SHAP values per m/z feature per pixel
    confidence: high
  Step 4: Biomarker candidate ranking
    tool: Custom Python script
    algorithm: mean absolute SHAP value aggregation across pixels of target class
    parameters: aggregation_mode=class-specific mean |abs(SHAP)|, top_k=100
    input: SHAP values matrix (m/z × pixel)
    output: ranked list of m/z features with biomarker potential scores
    confidence: medium
  Step 5: SHAP map generation
    tool: Custom Python visualization
    algorithm: spatial mapping of SHAP values per m/z feature
    parameters: colormap=RdBu_r, threshold_percentile=95
    input: SHAP values matrix + spatial coordinates
    output: SHAP maps (2D spatial heatmaps per m/z)
    confidence: low
