paper_title: Reproducible MS/MS library cleaning pipeline in matchms.

overall_reproducibility_score: 92/100

workflow_steps: 7
  Step 1: Input spectral library loading and format standardization
    tool: matchms (v0.19.0)
    algorithm: SpectrumLoader
    parameters: metadata_harmonization=True, ignore_unknown_adducts=True
    input: MSP, mgf, or mzML files containing MS/MS spectra with metadata
    output: matchms Spectrum objects with harmonized metadata
    confidence: high
  Step 2: Metadata curation and validation
    tool: matchms (v0.19.0)
    algorithm: MetadataValidator
    parameters: required_fields=['compound_name', 'smiles', 'inchi', 'precursor_mz', 'ionmode'], strict_smiles_validation=True, allow_missing_inchi=False
    input: harmonized Spectrum objects
    output: validated Spectrum objects with flagged/removed entries violating metadata rules
    confidence: high
  Step 3: Structure annotation validation via chemical consistency checks
    tool: rdkit
    algorithm: RDKit MolFromSmiles + sanitization + InChI round-trip validation
    parameters: sanitize=True, require_inchi_consistency=True, max_allowed_atom_count=100
    input: Spectra with SMILES and InChI annotations
    output: Spectra with validated or removed structure annotations
    confidence: high
  Step 4: Spectral quality filtering
    tool: matchms (v0.19.0)
    algorithm: SpectrumFilter
    parameters: min_peaks=5, min_intensity_fraction=0.01, remove_precursor_peak=True, precursor_tolerance Dalton=0.5
    input: validated Spectrum objects
    output: filtered Spectrum objects passing quality thresholds
    confidence: high
  Step 5: Adduct type inference and correction
    tool: matchms (v0.19.0)
    algorithm: AdductGuesser
    parameters: adducts_to_consider=['[M+H]+', '[M-H]-', '[M+Na]+', '[M+K]+', '[M+NH4]+', '[M-H2O+H]+'], mass_tolerance_ppm=10.0
    input: filtered Spectrum objects
    output: Spectrum objects with inferred/corrected adduct annotations
    confidence: high
  Step 6: Library deduplication and clustering
    tool: matchms (v0.19.0)
    algorithm: SpectrumSimilarity (Cosine)
    parameters: tolerance_ppm=10.0, n_decimals_mass=3, min_similarity_score=0.95, cluster_method=single_linkage
    input: adduct-corrected Spectrum objects
    output: deduplicated library with representative spectra per cluster
    confidence: medium
  Step 7: Logging, provenance tracking, and report generation
    tool: matchms (v0.19.0)
    algorithm: CleaningReportGenerator
    parameters: log_level=INFO, include_statistics=True, export_intermediate_results=False
    input: full cleaning pipeline state
    output: HTML/PDF cleaning report with metrics and audit trail
    confidence: high
