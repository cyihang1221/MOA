workflow_steps: 10
  Step 1: LC-MS spectra processing
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: asari
    parameters: auto_optimized=True
    input: raw .mzML or other open-format LC-MS spectra files (zipped)
    output: feature table (peak matrix)
    confidence: medium
  Step 2: LC-MS spectra processing (alternative)
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: centWave
    parameters: auto_optimized=True
    input: raw .mzML or other open-format LC-MS spectra files (zipped)
    output: feature table (peak matrix)
    confidence: medium
  Step 3: MS2 spectral deconvolution (DDA)
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: DecoID
    input: DDA MS2 spectra (MSP/MGF) linked to LC-MS features
    output: deconvolved MS2 spectra
    confidence: low
  Step 4: MS2 spectral deconvolution (SWATH-DIA)
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: DecoMetDIA (Rcpp/C++ reimplementation)
    input: SWATH-DIA MS2 spectra (zipped), with SWATH window design (auto-extracted or manually entered)
    output: deconvolved MS2 spectra
    confidence: low
  Step 5: Consensus spectrum generation
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: replicate-based consensus
    input: replicated deconvolved MS2 spectra
    output: consensus MS2 spectra
    confidence: low
  Step 6: MS2 database searching & annotation
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: dot product or spectral entropy similarity scoring
    input: consensus or single MS2 spectra + precursor m/z (if applicable)
    output: fragment-level annotated metabolite identifications
    confidence: medium
  Step 7: Statistical analysis for complex study designs
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: multivariate linear modeling (e.g., ANOVA, MANOVA, mixed-effects models)
    parameters: multiple_factors=True, covariate_adjustment=True
    input: feature table + rich phenotypic/metadata table
    output: differentially abundant metabolites, model diagnostics
    confidence: low
  Step 8: Causal analysis (metabolite–phenotype)
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: two-sample Mendelian randomization (2SMR)
    parameters: mGWAS_data_integration=True, GWAS_data_integration=True
    input: metabolite–phenotype associations + mGWAS + phenotype-GWAS summary statistics
    output: causal effect estimates (e.g., Wald ratio, IVW, MR-Egger)
    confidence: low
  Step 9: Dose-response analysis & benchmark dose (BMD) calculation
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: nonlinear curve fitting (e.g., Hill, log-logistic)
    parameters: confidence_interval=0.95
    input: dose–response concentration and effect data
    output: BMD, BMDL, best-fit model, dose-response curves
    confidence: low
  Step 10: Functional/pathway analysis
    tool: MetaboAnalyst 6.0 (v6.0)
    algorithm: over-representation analysis (ORA), pathway topology analysis
    parameters: species=≈130, database_updated=True
    input: annotated feature table (with HMDB/KEGG/ChEBI IDs)
    output: enriched pathways, metabolite sets, network visualizations
    confidence: medium
