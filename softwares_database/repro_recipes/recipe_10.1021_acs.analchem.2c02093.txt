paper_title: Rapid Approximate Subset-Based Spectra Prediction for Electron Ionization-Mass Spectrometry.

overall_reproducibility_score: 78/100

workflow_steps: 6
  Step 1: Subformula enumeration
    tool: custom_python
    algorithm: recursive Cartesian product over elemental subcounts
    parameters: elements=['H', 'C', 'O', 'N', 'F', 'S', 'P', 'Cl'], max_atom_count_per_element=20
    input: molecular graph (SMILES or InChI)
    output: list of chemical subformulae (count-encoded vectors)
    confidence: medium
  Step 2: Atom subset enumeration via bond-breaking
    tool: custom_python
    algorithm: recursive bond breaking up to depth d=3 + exhaustive hydrogen rearrangements
    parameters: bond_breaking_depth=3, include_hydrogen_rearrangements=True
    input: molecular graph (adjacency matrix + atom types)
    output: binary subset indicator matrix (NS × NA)
    confidence: high
  Step 3: Per-atom feature embedding
    tool: Graph Neural Network (GNN)
    algorithm: message-passing GNN with bond-order-aware adjacency matrix
    parameters: num_layers=None, feature_dim=F0 (unspecified), bond_order_encoding=[0, 1, 1.5, 2, 3]
    input: molecular graph (X0, A)
    output: per-atom embedding matrix Xd (NA × Fd)
    confidence: low
  Step 4: FormulaNet probability prediction
    tool: deep_learning_model
    algorithm: attention mechanism + MLP over formula embeddings and per-atom features
    parameters: attention_key=formula_embeddings, attention_query_value=Xd, softmax_scaling=linear_layer_from_Xd
    input: formula embeddings + Xd
    output: probability distribution over subformulae
    confidence: medium
  Step 5: SubsetNet probability prediction
    tool: deep_learning_model
    algorithm: subset masking + mean pooling + MLP
    parameters: masking_operation=mean_pooling_over_subset_atoms
    input: atom subset indicator matrix + Xd
    output: probability distribution over atom subsets
    confidence: medium
  Step 6: Spectrum synthesis from fragment probabilities
    tool: custom_spectral_simulation
    algorithm: isotopic m/z distribution summation weighted by fragment probability
    parameters: resolution=arbitrary, isotope_model=standard natural abundance
    input: fragment probability distribution + fragment identities (formulae or subsets)
    output: predicted EI-MS spectrum (m/z-intensity pairs)
    confidence: medium
