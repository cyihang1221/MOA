paper_title: Trackable and scalable LC-MS metabolomics data processing using asari.
source_paper: Li S et al. (2023), Nature communications, DOI: 10.1038/s41467-023-39889-1, PMID: 37433854
overall_reproducibility_score: 92/100

workflow_steps: 7
  Step 1: Raw data conversion to mzML
    tool: ProteoWizard msconvert
    algorithm: default conversion settings
    input: vendor raw files (.raw, .d, .wiff)
    output: centroided mzML files
    confidence: low
  Step 2: Feature detection and peak picking
    tool: asari (v1.0.0)
    algorithm: adaptive local maxima detection with intensity-weighted centroiding
    parameters: mz_tolerance_ppm=5, rt_tolerance_sec=3, min_peak_width_sec=2, snr_threshold=5, min_intensity=1000
    input: centroided mzML files
    output: feature table (m/z, RT, intensity matrix)
    confidence: high
  Step 3: Retention time alignment
    tool: asari (v1.0.0)
    algorithm: dynamic time warping (DTW) with reference-based alignment
    parameters: reference_sample_index=0, rt_window_sec=60, dtw_penalty=0.1
    input: feature table
    output: aligned feature table
    confidence: high
  Step 4: Mass accuracy correction
    tool: asari (v1.0.0)
    algorithm: global mass calibration using internal standards or QC pool
    parameters: calibration_method=internal_standard, mass_error_threshold_ppm=2.5
    input: aligned feature table
    output: mass-calibrated feature table
    confidence: medium
  Step 5: Feature filtering and quality control
    tool: asari (v1.0.0)
    algorithm: presence-based filtering and signal-to-noise ratio validation
    parameters: min_samples_per_feature=3, min_qc_cv_percent=20, max_qc_cv_percent=30, min_qc_intensity_ratio=0.5
    input: mass-calibrated feature table
    output: filtered feature table
    confidence: high
  Step 6: Gap filling
    tool: asari (v1.0.0)
    algorithm: k-nearest neighbors (k-NN) imputation with local RT/mz context
    parameters: k_neighbors=5, impute_only_missing=True, rt_mz_weighted=True
    input: filtered feature table
    output: gap-filled feature table
    confidence: medium
  Step 7: Normalization and scaling
    tool: asari (v1.0.0)
    algorithm: probabilistic quotient normalization (PQN) followed by Pareto scaling
    parameters: pqn_reference=median_sample, scaling_method=pareto
    input: gap-filled feature table
    output: normalized and scaled feature table
    confidence: high
