paper_title: Signature Mapping (SigMa): An efficient approach for processing complex human urine <sup>1</sup>H NMR metabolomics data.

overall_reproducibility_score: 30/100

workflow_steps: 5
  Step 1: Spectral segmentation
    tool: SigMa
    algorithm: signature-based spectral partitioning
    parameters: ss_region_width_ppm=0.02, sus_region_width_ppm=0.04, bin_width_ppm=0.01
    input: raw 1H NMR spectra (.fid or .jcamp)
    output: segmented spectral regions (SS, SUS, BINS)
    confidence: medium
  Step 2: Signature Signal (SS) detection
    tool: SigMa
    algorithm: library-matching with SigMa chemical shift library
    parameters: ss_match_tolerance_ppm=0.01, min_signal_to_noise_ratio=3
    input: SS spectral regions
    output: annotated SS metabolite list with chemical shifts and multiplicities
    confidence: medium
  Step 3: Signals of Unknown spin Systems (SUS) detection
    tool: SigMa
    algorithm: automatic peak picking + multivariate curve resolution (MCR)
    parameters: sus_peak_picking_threshold=adaptive SNR-based, mcr_max_iterations=100, mcr_convergence_threshold=1e-06
    input: SUS spectral regions
    output: quantified SUS metabolite concentrations
    confidence: low
  Step 4: Binning of unresolved regions (BINS)
    tool: SigMa
    algorithm: intensity binning
    parameters: bin_start_ppm=0.5, bin_end_ppm=9.5, bin_width_ppm=0.01
    input: unresolved inter-SS spectral regions
    output: binned intensity table (BINS matrix)
    confidence: high
  Step 5: Integration and table assembly
    tool: SigMa
    algorithm: concatenation of SS, SUS, and BINS quantifications
    parameters: integration_method=summed integral, normalization=probabilistic quotient normalization (PQN)
    input: SS concentrations, SUS concentrations, BINS intensities
    output: final metabolite table (SS + SUS + BINS features)
    confidence: medium
