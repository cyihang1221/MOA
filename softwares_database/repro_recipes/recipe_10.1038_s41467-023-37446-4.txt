paper_title: MS2Query: reliable and scalable MS<sup>2</sup> mass spectra-based analogue search.
source_paper: de Jonge NF et al. (2023), Nature communications, DOI: 10.1038/s41467-023-37446-4, PMID: 36990978
overall_reproducibility_score: 92/100

workflow_steps: 7
  Step 1: Spectral data cleaning and preprocessing
    tool: matchms
    algorithm: intensity normalization, peak filtering, InChIKey imputation
    parameters: max_mz=1000, min_intensity_percent=0.1, max_peaks=500, min_peaks_per_spectrum=3, fragment_mass_tolerance_da=0.05
    input: raw .mgf files from GNPS
    output: cleaned, normalized, metadata-standardized spectra with imputed InChIKeys
    confidence: high
  Step 2: Chemical fingerprint generation and structural similarity calculation
    tool: RDKit
    algorithm: Daylight fingerprint (2048-bit) + Tanimoto coefficient
    parameters: fingerprint_type=daylight, n_bits=2048, inchikey_prefix_length=14
    input: SMILES (sanitized) and InChIKeys from annotated spectra
    output: pairwise Tanimoto similarity matrix for all unique 2D structures
    confidence: high
  Step 3: Spec2Vec model training
    tool: Spec2Vec
    algorithm: word2vec-based spectral embedding
    parameters: epochs=30, binning_decimals=2, use_unannotated_spectra=True
    input: all GNPS spectra (annotated + unannotated)
    output: precomputed Spec2Vec embeddings for all library spectra
    confidence: high
  Step 4: MS2Deepscore model training
    tool: MS2Deepscore
    algorithm: Siamese neural network with cosine similarity on spectral embeddings
    parameters: training_data=fully annotated GNPS spectra only, settings_from_publication=True
    input: fully annotated GNPS spectra (with SMILES, InChI, InChIKey)
    output: precomputed MS2Deepscore embeddings and scoring function
    confidence: medium
  Step 5: Training spectrum pair generation for random forest
    tool: custom Python script (scikit-learn + matchms)
    algorithm: top-100 MS2Deepscore pairing per query spectrum
    parameters: fraction_of_inchikeys_for_training=0.025, fraction_of_remaining_spectra=0.025, top_k_ms2deepscore_pairs=100
    input: MS2Deepscore scores + Tanimoto labels
    output: labeled spectrum pairs with Tanimoto score as target
    confidence: high
  Step 6: Random forest model training
    tool: scikit-learn
    algorithm: impurity-based (Gini) random forest regression
    parameters: n_estimators=250, max_depth=5, loss_function=mean_squared_error, feature_importance_method=Gini importance
    input: 5-feature vectors (Spec2Vec sim, precursor m/z, Δm/z, avg_MS2Deepscore_10sim, avg_Tanimoto_10sim) + Tanimoto label
    output: trained random forest model for analogue ranking
    confidence: high
  Step 7: Analogue search execution
    tool: MS2Query (v1.0.0 (implied by publication date))
    algorithm: two-stage ranking: MS2Deepscore prefiltering → RF re-ranking
    parameters: preselection_count=2000, chemical_similarity_neighbors=10, precursor_mz_tolerance_da=100.0, exact_match_mz_tolerance_da=0.25
    input: query spectrum (.mgf) + precursor m/z + precomputed library embeddings & SQLite metadata
    output: ranked list of analogues and exact matches with scores
    confidence: high
