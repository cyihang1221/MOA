workflow_steps: 3
  Step 1: Ion mobility arrival time calibration
    tool: MOCCal
    algorithm: class-specific power law calibration with TWIM-CCS correction using DTIM-CCS regression residuals
    parameters: calibrant_classes=['small molecules', 'lipids', 'peptides'], calibration_curves_used=3, residual_threshold_Å2=None, gas=N2, ccs_unit=Å²
    input: TWIM arrival times, m/z values, class annotations (or inferred classes)
    output: calibrated TWCCS_N2 values with class assignment and residual error (Å²)
    confidence: medium
  Step 2: Multiomic feature classification
    tool: MOCCal
    algorithm: Trendline-based biomolecule class assignment using m/z vs CCS (or arrival time) organization
    input: m/z and calibrated or uncalibrated arrival time/CCS data
    output: feature-class assignments (e.g., lipid, peptide, metabolite)
    confidence: medium
  Step 3: CCS accuracy assessment
    tool: MOCCal
    algorithm: Residual calculation between TWCCS_N2 and reference DTCCS_N2 vs m/z regression lines
    parameters: reference_method=drift tube IM (DTIM), regression_type=power law
    input: TWCCS_N2 values, DTCCS_N2 reference regression parameters
    output: residual error (Å²) per feature; optimal calibration curve selection
    confidence: high
