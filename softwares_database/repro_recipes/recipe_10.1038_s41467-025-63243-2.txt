workflow_steps: 7

    algorithm: msconvert
    parameters: to_mzXML=True, to_mgf=True, filter=peakPicking
    input: vendor .raw files
    output: .mzXML (MS1) and .mgf (MS/MS) files
    confidence: high
  Step 2: Parameter optimization for peak detection
    tool: AutoTuner (v1.4.0)
    algorithm: grid search + signal-to-noise optimization
    input: .mzXML files (noninfusion controls)
    output: optimized XCMS parameters (e.g., ppm tolerance, peak width, snthresh)
    confidence: medium
  Step 3: Feature detection, retention time correction, and peak alignment
    tool: XCMS (v3.12.0)
    algorithm: centWave (for peak detection), obiwarp (for retention time correction), and matchFeatures (for alignment)
    parameters: ppm=None, peakwidth=None, snthresh=None, prefilter=3, mzdiff=-0.001, noise=1000, sleep=0.01
    input: optimized .mzXML files (noninfusion group)
    output: aligned feature table (m/z, RT, intensity matrix)
    confidence: low
  Step 4: Metabolite annotation (HILIC mode)
    tool: metID (v1.2.19)
    algorithm: MS2 spectral matching + retention time prediction
    parameters: lc_mode=HILIC, database=HMDB, METLIN, MassBank
    input: XCMS feature table + .mgf MS/MS files
    output: annotated metabolite table (Level 2 & 3 confidence)
    confidence: high
  Step 5: Metabolite annotation (RP mode)
    tool: metID (v1.2.19)
    algorithm: MS1 database matching (exact mass + RT filtering)
    parameters: lc_mode=RP, database=HMDB, METLIN, MassBank
    input: XCMS feature table
    output: annotated metabolite table (Level 2 & 3 confidence)
    confidence: high
  Step 6: Metabolite annotation (HILIC mode, alternative)
    tool: MetDNA2 (v1.4.1)
    algorithm: MS2 spectral networking + in silico fragmentation
    parameters: lc_mode=HILIC, ionization_mode=positive/negative
    input: XCMS feature table + .mgf MS/MS files
    output: annotated metabolite table (Level 2 & 3 confidence)
    confidence: medium
  Step 7: Isotopic enrichment quantification
    tool: MSITracer
    algorithm: spatial isotope deep tracing (SIDT): differential isotopologue distribution analysis across tissue regions
    parameters: isotope_label=U-13C, background_subtraction=noninfusion control, spatial_resolution=100 μm × 76 μm (tissue), 500 μm × 270 μm (WBA), mz_tolerance_ppm=3
    input: AFADESI-MSI ion images + LC-MS isotopologue intensities
    output: spatially resolved 13C-enrichment maps and inter-tissue metabolic flux estimates
    confidence: low
