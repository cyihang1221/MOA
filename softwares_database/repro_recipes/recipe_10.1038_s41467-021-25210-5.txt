workflow_steps: 7
  Step 1: Sample preparation and metabolite extraction
    tool: HILIC sample buffer (ACN:MeOH:FA 75:25:0.2 v/v:v)
    algorithm: protein precipitation
    parameters: plasma_volume_ul=10, buffer_volume_ul=90, centrifugation_rpm=14000, centrifugation_min=20
    input: frozen plasma samples
    output: metabolite-containing supernatant
    confidence: high
  Step 2: LC-MS acquisition
    tool: Agilent 1260 Infinity HPLC + Waters Atlantis Silica HILIC column + Sciex QTRAP 5500
    algorithm: scheduled MRM
    parameters: gradient_duration_min=25, mobile_phase_A=0.1% formic acid + 10 mM ammonium formate, mobile_phase_B=0.1% formic acid in acetonitrile, batch_duration_h=40, total_batches=15, samples_per_batch=88
    input: metabolite-containing supernatant
    output: raw .wiff files (Sciex native format)
    confidence: high
  Step 3: Peak integration and quantification
    tool: MultiQuant (v3.03)
    algorithm: area under the curve (AUC) with manual curation
    parameters: integration_method=calibrated peak AUC, QC_removal_criteria=outlier removal from first three consecutive QC samples
    input: raw .wiff files
    output: quantified signal matrix (metabolites × samples)
    confidence: high
  Step 4: Data pre-processing
    tool: R (DMwR2, stats, MASS) (vR 4.0.3, DMwR2 0.0.2, MASS 7.3-53.1)
    algorithm: log2 transformation + missing value imputation + low-abundance filtering
    parameters: log_transform=log2, presence_threshold=0.5, imputation_method=k-nearest neighbour, imputation_k=None, outlier_removal=first three QC samples per batch
    input: quantified signal matrix
    output: pre-processed feature table (log2-transformed, filtered, imputed)
    confidence: medium
  Step 5: Within-batch signal drift correction
    tool: hRUV R package
    algorithm: multiple options: ratio-to-QC, loess (span=0.75), robust linear model (rlm, maxit=100), QC-only loess/rlm, or RUV with replicates
    parameters: loess_span=0.75, rlm_maxit=100, ruv_k=5, ruv_negative_controls=all metabolites (default) or data-driven selection
    input: pre-processed feature table
    output: within-batch normalised feature table
    confidence: high
  Step 6: Hierarchical inter-batch correction
    tool: hRUV R package
    algorithm: balanced tree or concatenation RUV
    parameters: batch_correction_strategy=['balanced_tree', 'concatenation'], ruv_k=5, tree_depth=log2(15) ≈ 4 layers for balanced tree, n_adjustments=14 for concatenation
    input: within-batch normalised feature table
    output: fully hRUV-normalised feature table
    confidence: high
  Step 7: Data-driven negative control selection (optional)
    tool: hRUV R package
    algorithm: ranking by p-value for hypertension response variable per batch
    parameters: response_variable=hypertension, selection_criterion=non-differentially expressed metabolites
    input: within-batch normalised feature table + metadata
    output: subset of negative control metabolites for RUV
    confidence: low
