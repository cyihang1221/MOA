overall_reproducibility_score: 78/100

workflow_steps: 7
  Step 1: Resolution-based isotopic envelope calculation
    tool: HERMES
    algorithm: centroidization with resolution-aware peak discrimination
    parameters: mass_tolerance_ppm=0.27
    input: raw LC/MS1 profile data
    output: centroided MS1 signals with distinguishable isotopologues
    confidence: medium
  Step 2: Blank subtraction
    tool: HERMES
    algorithm: artificial neural network (ANN)
    input: centroided MS1 signals
    output: blank-subtracted SOI candidates
    confidence: low
  Step 3: Adduct and isotopologue grouping
    tool: HERMES
    algorithm: elution profile similarity clustering
    input: blank-subtracted SOI candidates
    output: grouped adducts and isotopologues
    confidence: low
  Step 4: In-source fragment (ISF) annotation
    tool: HERMES
    algorithm: matching against low-energy MS2 reference data
    input: grouped adducts/isotopologues
    output: ISF-filtered SOI list
    confidence: low
  Step 5: Continuous MS2 deconvolution
    tool: HERMES
    algorithm: fragment elution profile grouping + greedy clustering
    input: continuous MS2 scans
    output: curated, co-elution-resolved MS2 spectra per SOI
    confidence: medium
  Step 6: Isotopic fidelity scoring and filtering
    tool: HERMES
    algorithm: characteristic isotopologue presence/absence detection (e.g., ³⁷Cl, ⁴¹K)
    parameters: isotopic_fidelity_threshold=None
    input: SOI list with isotopic patterns
    output: filtered high-fidelity SOI inclusion list
    confidence: medium
  Step 7: MS2 spectral matching and metabolite identification
    tool: HERMES
    algorithm: cosine similarity scoring + database matching
    parameters: cosine_score_threshold=0.8, mzCloud_match_threshold=90, mzCloud_confidence_threshold=30
    input: curated MS2 spectra
    output: identified metabolites (compound hits)
    confidence: high
