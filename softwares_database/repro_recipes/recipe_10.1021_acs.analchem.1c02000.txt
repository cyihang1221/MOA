paper_title: Assessment of XCMS Optimization Methods with Machine-Learning Performance.
source_paper: Lassen J et al. (2021), Analytical chemistry, DOI: 10.1021/acs.analchem.1c02000, PMID: 34585906
overall_reproducibility_score: 92/100

workflow_steps: 7
  Step 1: Raw data conversion to mzML
    tool: ProteoWizard msconvert
    algorithm: default conversion settings
    input: vendor raw files (.raw, .d)
    output: mzML files
    confidence: medium
  Step 2: Feature detection and alignment
    tool: XCMS (v3.12.0)
    algorithm: centWave
    parameters: ppm=5, peakwidth=[10, 60], snthresh=10, prefilter=[3, 100], noise=0, verbose.columns=False
    input: mzML files
    output: feature table (ion matrix)
    confidence: high
  Step 3: Retention time correction
    tool: XCMS (v3.12.0)
    algorithm: loess
    parameters: span=0.2, plottype=none
    input: feature table with retention time shifts
    output: retention time-aligned feature table
    confidence: high
  Step 4: Peak filling and missing value imputation
    tool: XCMS (v3.12.0)
    algorithm: fillPeaks
    parameters: minfrac=0.5, max=3
    input: aligned feature table
    output: gap-filled feature table
    confidence: high
  Step 5: Quality assessment via machine learning
    tool: R (caret, randomForest, e1071)
    algorithm: random forest classification
    parameters: ntree=500, mtry=sqrt(p), repeatedcv_folds=5, repeats=10
    input: normalized feature table
    output: classification accuracy (AUC, balanced accuracy)
    confidence: high
  Step 6: XCMS parameter optimization
    tool: Autotuner (v1.4.0)
    algorithm: genetic algorithm
    parameters: n_iterations=50, population_size=20, mutation_rate=0.1
    input: mzML files and class labels
    output: optimized XCMS parameters
    confidence: high
  Step 7: XCMS parameter optimization
    tool: IPO (isotopologue parameter optimization) (v1.18.0)
    algorithm: isotopologue-based scoring
    parameters: minfrac=0.5, maxiso=3, mzdiff=0.01
    input: mzML files and isotopologue patterns
    output: optimized XCMS parameters
    confidence: high
