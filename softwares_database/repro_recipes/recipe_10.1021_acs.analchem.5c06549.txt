workflow_steps: 12
  Step 1: CentWave peak detection
    tool: xcms R package v. 4.3.4
    algorithm: centWave
    parameters: ppm=5, peakwidth=[4,30], peakwidth=[1,25], snthresh=10, integrate=TRUE, extendLengthMSW=TRUE, firstBaselineCheck=TRUE
    input: raw UHPLC-HRMS .raw files (Q-Exactive, 100,000 FWHM, 70–700 Da)
    output: chromatographic peaks (m/z–RT pairs with intensity)
    confidence: high
  Step 2: Retention time alignment (ENVIRONAGE)
    tool: xcms R package v. 4.3.4
    algorithm: OBI-Warp
    parameters: binSize=0.01, centerSample=1
    input: centWave-detected peaks from 10 sample runs
    output: aligned RTs across samples
    confidence: high
  Step 3: Retention time alignment (ESCCAPE)
    tool: xcms R package v. 4.3.4
    algorithm: PeakGroups
    parameters: anchor_peaks=[13C3]-M1-G,[13C,15N2]-Cro-dG,O6-[d3]-Me-dG
    input: centWave-detected peaks from 10 sample runs
    output: aligned RTs across samples
    confidence: high
  Step 4: Feature grouping
    tool: xcms R package v. 4.3.4
    algorithm: PeakDensity
    parameters: bw=2, minFraction=0.1, minFraction=0.05, ppm=5, binSize=0.01
    input: aligned peaks
    output: features (consensus m/z–RT clusters across samples)
    confidence: high
  Step 5: Gap filling
    tool: xcms R package v. 4.3.4
    algorithm: fillChromPeaks()
    parameters: not_reported
    input: original raw files and feature list
    output: integrated intensities for missing features
    confidence: medium
  Step 6: Missing value filtering
    tool: R v. 4.4.1
    algorithm: feature removal
    parameters: missing_threshold=50%
    input: feature matrix with missing values
    output: filtered feature matrix
    confidence: high
  Step 7: Missing value imputation
    tool: R v. 4.4.1
    algorithm: uniform sampling imputation
    parameters: range=[0.5*min_value, min_value]
    input: filtered feature matrix
    output: imputed feature matrix
    confidence: high
  Step 8: QC-based robust LOESS smoothing (QC-RLSC)
    tool: fANCOVA R package
    algorithm: LOESS with GCV
    parameters: loess.as_gcv=TRUE
    input: imputed feature matrix and iQC/eQC runs
    output: QC-RLSC normalized feature matrix
    confidence: high
  Step 9: Within-batch normalization + batch alignment
    tool: R v. 4.4.1
    algorithm: mean response alignment
    parameters: batch_definition=column_1&2_vs_column_3, alignment_method=mean_response
    input: QC-RLSC normalized data per batch
    output: batch-corrected feature matrix
    confidence: high
  Step 10: RSD* and D-ratio filtering
    tool: MetaboCoreUtils R package
    algorithm: rsd() and rowDratio()
    parameters: mad=TRUE, RSD*_threshold=0.2, RSD*_threshold=0.3, D-ratio_threshold=0.4, D-ratio_threshold=0.5
    input: normalized feature matrix, ISTDs, technical replicates, QCs
    output: final retained features for downstream analysis
    confidence: high
  Step 11: PCA evaluation
    tool: R v. 4.4.1
    algorithm: principal component analysis
    parameters: log2_transform=TRUE, center=TRUE, scale=TRUE
    input: final retained features
    output: PCA score plots for QC/sample clustering assessment
    confidence: high
  Step 12: DNA concentration correction
    tool: R v. 4.4.1
    algorithm: linear scaling
    parameters: correction_factor=DNA_concentration / mean_DNA_concentration
    input: final retained features and sample DNA concentrations (ng/μL)
    output: DNA-concentration-normalized feature matrix
    confidence: high
