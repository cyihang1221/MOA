workflow_steps: 10
  Step 1: Data conversion and preprocessing
    tool: OpenMS
    algorithm: FileConverter, PeakPickerHiRes
    parameters: peak_width=0.1, signal_to_noise=3
    input: raw vendor files (.raw, .d, .mzXML)
    output: centroided .mzML files
    confidence: medium
  Step 2: Feature detection and alignment
    tool: OpenMS
    algorithm: FeatureFinderMetabo
    parameters: mass_error_ppm=5, rt_tolerance=30, intensity_threshold=1000
    input: centroided .mzML files
    output: feature table (consensusXML)
    confidence: high
  Step 3: Retention time correction
    tool: OpenMS
    algorithm: MapAlignerIdentification
    parameters: reference_file=first_sample.mzML, method=lowess
    input: consensusXML with aligned features
    output: RT-corrected consensusXML
    confidence: medium
  Step 4: Gap filling
    tool: OpenMS
    algorithm: FeatureFinderMultiplex
    parameters: use_identifications=True, min_rt_intensity=100
    input: RT-corrected consensusXML
    output: gap-filled feature table
    confidence: medium
  Step 5: Molecular formula prediction
    tool: SIRIUS
    algorithm: CSI:FingerID + CANOPUS
    parameters: max_adducts=5, ppm_tolerance=10, isotope_count=3
    input: feature table with MS/MS spectra
    output: molecular formula candidates with scores
    confidence: high
  Step 6: Structural annotation
    tool: SIRIUS
    algorithm: CSI:FingerID
    parameters: fingerprint_similarity_threshold=0.3
    input: molecular formula candidates + MS/MS spectra
    output: ranked structure candidates
    confidence: high
  Step 7: Dereplication and spectral matching
    tool: GNPS
    algorithm: Feature-Based Molecular Networking (FBMN)
    parameters: cosine_score_threshold=0.7, min_matched_peaks=6, max_mz_diff=0.02
    input: feature table + MS/MS spectra
    output: molecular families and annotated nodes
    confidence: high
  Step 8: Ion Identity Molecular Networking (IIMN)
    tool: GNPS
    algorithm: IIMN
    parameters: adduct_tolerance_ppm=5, ionization_mode=positive
    input: feature table + adduct annotations
    output: ion identity networks
    confidence: medium
  Step 9: Statistical analysis and biomarker selection
    tool: MetaboAnalyst
    algorithm: ANOVA + FDR correction, PLS-DA, ROC analysis
    parameters: p_value_cutoff=0.05, fdr_method=BH, n_components=2, auc_threshold=0.8
    input: normalized feature table
    output: differential features, biomarkers, model metrics
    confidence: medium
  Step 10: Pathway enrichment and biological interpretation
    tool: MetaboAnalyst
    algorithm: Hypergeometric test + topology analysis
    parameters: database=HMDB, organism=Homo sapiens
    input: annotated differential features
    output: enriched pathways, impact scores
    confidence: low
