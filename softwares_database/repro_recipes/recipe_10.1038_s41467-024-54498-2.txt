paper_title: Impact of easing COVID-19 restrictions on antibiotic usage in Eastern China using wastewater-based epidemiology.

workflow_steps: 10
  Step 1: Sample collection
    tool: JH-8000D auto-sampler
    parameters: sampling_interval_minutes=15, duration_hours=24, sample_volume_per_interval_mL=10, total_samples_per_campaign=70
    input: raw wastewater influent
    output: 24-hour composite wastewater samples
    confidence: high
  Step 2: Internal standard spiking
    tool: manual pipetting
    parameters: istd_volume_uL=50, istd_concentration_mg_L=1.0, istd_mass_ng=50
    input: composite wastewater samples
    output: ISTD-spiked wastewater samples
    confidence: high
  Step 3: Solid-phase extraction (SPE)
    tool: Oasis HLB cartridges
    parameters: cartridge_capacity_mg=60, filtration_pore_size_um=0.7, load_flow_rate_mL_min=5, conditioning_solvents=['2 mL MeOH', '2 mL HPLC water'], elution_volume_mL=4, elution_solvent=MeOH
    input: ISTD-spiked wastewater samples
    output: SPE eluate
    confidence: high
  Step 4: Sample concentration and reconstitution
    tool: nitrogen evaporator (LabTech)
    parameters: evaporation_temperature_C=40, reconstitution_volume_mL=0.5, reconstitution_solvent_ratio=4:1 H2O:MeOH
    input: SPE eluate
    output: reconstituted extract for injection
    confidence: high
  Step 5: Chromatographic separation
    tool: ACQUITY UPLC™ system with HSS T3 column
    parameters: column_dimensions_mm=2.1 × 100, particle_size_um=1.8, gradient_duration_min=11.5, mobile_phase_A=2:3 methanol:acetonitrile, mobile_phase_B=0.2% formic acid in water
    input: reconstituted extract
    output: chromatographically separated analytes
    confidence: medium
  Step 6: Quantitative detection via MRM
    tool: ABSCIEX QTRAP 6500+
    algorithm: multiple reaction monitoring (MRM)
    parameters: transitions_per_analyte=2, transitions_per_ISTD=1, mdls_ng_L=[0.004, 0.296], recovery_range_percent=[70.0, 120.8]
    input: chromatographically separated analytes
    output: quantified antibiotic concentrations (ng/L)
    confidence: high
  Step 7: WBE load and intake calculation
    tool: Microsoft Excel (v2019)
    algorithm: population-normalised daily load/intake formulas
    parameters: equation_1=DL = C × F, equation_2=PNDL = (C × F) / (PE × 10^3), equation_3=PNDI = PNDL × CF
    input: antibiotic concentrations (ng/L), flow rate (L/day), population estimate (PE), correction factors (CF)
    output: PNDL (mg/day/1000 inh), PNDI (mg/day/1000 inh)
    confidence: high
  Step 8: Statistical comparison (temporal change)
    tool: SPSS Statistics (v26.0)
    algorithm: Wilcoxon rank-sum test
    parameters: alpha=0.05, two_sided=True, bonferroni_adjustment=True
    input: PNDL/PNDI values for 2021 vs 2023
    output: p-values, fold-changes, significance flags
    confidence: high
  Step 9: Temporal variability analysis
    tool: SPSS Statistics (v26.0)
    algorithm: ANOVA
    parameters: alpha=0.05, factor=day (1–7)
    input: daily antibiotic concentrations across 7 days
    output: F-statistic, p-value, group means
    confidence: high
  Step 10: Correlation analysis
    tool: GraphPad Prism (v9.0.0)
    algorithm: Pearson correlation
    parameters: alpha=0.05, bonferroni_adjustment=True
    input: antibiotic PNDI or concentration matrices
    output: correlation matrix (r values), significance flags
    confidence: high
