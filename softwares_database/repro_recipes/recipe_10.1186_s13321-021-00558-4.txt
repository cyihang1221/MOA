paper_title: MS2DeepScore: a novel deep learning similarity measure to compare tandem mass spectra

overall_reproducibility_score: 92/100

workflow_steps: 11
  Step 1: MS/MS data acquisition and metadata cleaning
    tool: matchms (v0.8.2)
    algorithm: metadata standardization
    parameters: ion_mode=positive, min_peaks=5, mz_range_min=10.0, mz_range_max=1000.0, inchikey_length=14
    input: raw GNPS .mzML/.mgf spectra with metadata
    output: cleaned spectrum objects with InChIKey, SMILES, InChI
    confidence: high
  Step 2: Compound annotation enrichment
    tool: pubchempy
    algorithm: PubChem API lookup
    input: spectra missing InChI/SMILES
    output: fully annotated spectra (InChIKey + SMILES/InChI)
    confidence: medium
  Step 3: Spectral preprocessing and binning
    tool: matchms (v0.8.2)
    algorithm: intensity-based peak filtering and fixed-width binning
    parameters: intensity_threshold_percent=0.1, max_peaks=1000, intensity_transformation=square_root, mz_min=10.0, mz_max=1000.0, n_bins=10000, bins_used=9948
    input: cleaned spectrum objects
    output: binned spectral vectors (9948-dim)
    confidence: high
  Step 4: Structural similarity labeling
    tool: RDKit
    algorithm: Tanimoto similarity on Daylight fingerprints (2048 bits)
    parameters: fingerprint_type=Daylight, n_bits=2048, similarity_metric=Tanimoto
    input: SMILES/InChI
    output: 15,062 × 15,062 symmetric Tanimoto similarity matrix
    confidence: high
  Step 5: Dataset splitting
    tool: custom Python
    algorithm: stratified by InChIKey
    parameters: train_inchikeys=14062, validation_inchikeys=500, test_inchikeys=500, validation_spectra=3597, test_spectra=3601, random_seed=fixed
    input: annotated spectra + Tanimoto matrix
    output: train/validation/test spectrum sets with labels
    confidence: high
  Step 6: Data generation with balanced sampling
    tool: MS2DeepScore DataGeneratorAllSpectrums
    algorithm: bin-stratified random pairing with adaptive bin widening
    parameters: n_bins=10, bin_widening_step=0.1, max_bin_widening=0.5
    input: training spectra + Tanimoto matrix
    output: balanced batches of spectrum pairs with Tanimoto labels
    confidence: high
  Step 7: Data augmentation
    tool: MS2DeepScore
    algorithm: stochastic spectral perturbation
    parameters: low_intensity_removal_percent_range=[0, 20], intensity_jitter_percent_range=[-40, 40], new_peak_addition_count_range=[0, 10], new_peak_intensity_range=[0, 0.01], intensity_threshold_for_removal=0.4
    input: binned spectral vectors
    output: augmented binned spectral vectors
    confidence: high
  Step 8: Siamese neural network training
    tool: TensorFlow/Keras
    algorithm: Siamese network with cosine head
    parameters: base_network_architecture=['Dense(500)', 'Dense(500)', 'Dense(200)'], embedding_dim=200, optimizer=Adam, learning_rate=0.001, batch_size=32, loss_function=MSE, early_stopping_patience=5, l1_regularization=1e-06, l2_regularization=1e-06, dropout_rate=0.2, batch_normalization=True
    input: augmented spectrum pairs + Tanimoto labels
    output: trained Siamese model (base + head)
    confidence: high
  Step 9: Monte-Carlo Dropout inference and uncertainty estimation
    tool: MS2DeepScore
    algorithm: MC-Dropout ensemble (N=10)
    parameters: mc_dropout_enabled_layers=['dense_2', 'dense_3'], n_samples=10, uncertainty_metric=interquartile_range
    input: single binned spectrum or spectrum pair
    output: median similarity score + IQR uncertainty estimate
    confidence: high
  Step 10: Spectral embedding generation and visualization
    tool: scikit-learn
    algorithm: t-SNE
    parameters: n_components=2, metric=cosine, perplexity=100, learning_rate=200, n_iter=1000, random_state=None
    input: 200-dimensional spectral embeddings
    output: 2D t-SNE coordinates
    confidence: high
  Step 11: Performance evaluation (precision/recall)
    tool: custom Python
    algorithm: threshold sweep over similarity scores
    parameters: structural_similarity_threshold=0.6, score_threshold_range=[0, 0.99]
    input: predicted similarity scores + true Tanimoto labels
    output: precision-recall curve
    confidence: high
