workflow_steps: 4
  Step 1: High-resolution MALDI-MSI data acquisition
    tool: Multiple vendor mass spectrometers
    parameters: ionization_mode=positive, matrix=2,5-dihydroxybenzoic acid (DHB), mass_resolution=high_mass_resolution
    input: tissue sections
    output: raw .imzML or vendor-specific MSI datasets
    confidence: medium
  Step 2: Adduct detection via co-localization and mass accuracy
    tool: mass2adduct
    algorithm: co-localization-based adduct assignment
    parameters: mass_tolerance_ppm=5, spatial_correlation_threshold=0.7, minimum_peak_intensity=user-defined, adduct_templates=['[M + (DHB-H2O) + H]+', '[M + DHB + Na]+']
    input: raw MSI datasets (.imzML or vendor formats)
    output: putative metabolite-matrix adduct table with spatial maps
    confidence: high
  Step 3: Adduct quantification and abundance estimation
    tool: mass2adduct
    algorithm: intensity-based relative quantification
    parameters: normalization_method=total_ion_current, abundance_threshold=relative_to_parent_ion
    input: adduct-annotated peak lists with spatial intensity matrices
    output: adduct abundance matrix (per dataset)
    confidence: high
  Step 4: Cross-platform adduct frequency analysis
    tool: R
    algorithm: descriptive statistics across 20 datasets
    parameters: datasets_count=20, sample_types=diverse biological tissues, vendor_platforms=multiple
    input: adduct abundance matrices from step 3
    output: summary statistics on adduct prevalence (e.g., up to one third of peaks)
    confidence: medium
