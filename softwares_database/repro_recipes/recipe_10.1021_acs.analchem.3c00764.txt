overall_reproducibility_score: 87/100

workflow_steps: 5
  Step 1: Peak detection (initial)
    tool: XCMS
    algorithm: centWave
    parameters: ppm=5, snthr=10, prefilter=[5, 100]
    input: raw .mzML files
    output: feature table with putative peaks
    confidence: medium
  Step 2: Peak representation learning
    tool: PeakDetective
    algorithm: unsupervised autoencoder
    parameters: latent_dim=32, input_shape=[64, 1], epochs=50
    input: extracted peak chromatograms (intensity vs. retention time)
    output: low-dimensional latent representations of peaks
    confidence: high
  Step 3: Artifact classification via active learning
    tool: PeakDetective
    algorithm: SVM classifier with active learning loop
    parameters: initial_labeled_count=50, max_labeled_count=100, query_strategy=uncertainty_sampling, batch_size=10
    input: latent representations + user-labeled peaks (artifacts vs. true)
    output: binary artifact/true peak classification per feature
    confidence: high
  Step 4: Curated feature table generation
    tool: PeakDetective
    algorithm: filtering by classifier confidence threshold
    parameters: confidence_threshold=0.8
    input: original feature table + classifier predictions
    output: curated feature table (artifacts removed)
    confidence: high
  Step 5: Re-trained peak detection
    tool: PeakDetective
    algorithm: classifier-guided peak detection
    parameters: detection_sensitivity=high, selectivity_constraint=classifier_confidence > 0.9
    input: raw .mzML files
    output: de novo feature table with high sensitivity & selectivity
    confidence: medium
