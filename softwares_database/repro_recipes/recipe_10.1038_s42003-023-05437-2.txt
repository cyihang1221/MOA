paper_title: Achieving quantitative reproducibility in label-free multisite DIA experiments through multirun alignment.

overall_reproducibility_score: 87/100

workflow_steps: 12
  Step 1: Raw data conversion to mzML
    tool: MSConvert
    algorithm: no peak-picking
    parameters: docker_image=chambm/pwiz-skyline-i-agree-to-the-vendor-licenses:c30f8e5beb5f, compression=['linear m/z', 'positive integer intensity']
    input: vendor-specific spectra files (.raw, .d, etc.)
    output: mzML files
    confidence: high
  Step 2: DIA library preparation
    tool: custom curation pipeline
    algorithm: library harmonization
    parameters: rt_normalization=average NormalizedRetentionTime across charge states, rt_filter=remove peptides with |ΔNormalizedRetentionTime| > 4 across charge states, sequence_validation=Uniprot (Dec 2021) matching + manual correction, iRT_AQUA_addition=['11 iRTs', '30 AQUA peptides']
    input: peptide spectral libraries (e.g., .tsv, .splib)
    output: curated DIA library (.splib or .tsv)
    confidence: medium
  Step 3: Peak picking and XIC extraction
    tool: OpenSWATH
    algorithm: targeted extraction using DIA library
    parameters: RT_extraction_window_s=600, extra_RT_window_s=50, min_upper_edge_dist=1, MS2_extraction_window_ppm=75, MS1_extraction_window_ppm=35, DIA_extraction_window_ppm=75, mass_correction=ppm quadratic regression, background_subtraction=vertical_division_min, scoring=['mutual_information', 'MS1_scoring']
    input: mzML files + DIA library
    output: feature files (.osw) and chromatogram files (.chrom.mzML)
    confidence: high
  Step 4: Chromatogram compression to sqMass
    tool: OpenSwathMzMLFileCacher
    algorithm: lossless chromatogram serialization
    parameters: lossy_compression=False
    input: .chrom.mzML
    output: .chrom.sqMass
    confidence: high
  Step 5: Feature scoring and FDR control
    tool: PyProphet (v2.1.10)
    algorithm: XGBoost classifier
    parameters: level=ms1ms2, initial_FDR=0.01, integration_FDR=0.05, mscore_threshold=0.025, q_value_threshold=0.025
    input: .osw files
    output: FDR-calibrated feature table with mscore and q-value
    confidence: high
  Step 6: Multirun chromatogram alignment
    tool: DIAlignR (v>2.3)
    algorithm: ['Star (reference-based)', 'MST (minimum spanning tree)', 'Progressive (reference-free)']
    parameters: transitionIntensity=True, hardConstrain=True, maxFdrQuery=0.05, alignedFDR1=0.05, alignedFDR2=0.05, fractionation=10, signal_integration_within_run=True
    input: sqMass chromatograms + PyProphet-scored features
    output: aligned data matrix (peptide-level intensities across runs)
    confidence: high
  Step 7: Alternative alignment for comparison
    tool: TRIC (v0.11.0)
    algorithm: MST-based RT correction with LOWESS
    parameters: readmethod=cminimal, realign_method=lowess_cython, mst_Stdev_multiplier=4.0, mst_useRTCorrection=True
    input: OpenSWATH .osw files
    output: TRIC-aligned data matrix
    confidence: high
  Step 8: Data normalization and transformation
    tool: R base + custom scripts
    algorithm: median normalization + log2 transformation
    parameters: normalization_strategy=median-normalize per run, transformation=log2
    input: aligned peptide intensity matrix
    output: normalized log2-intensity matrix
    confidence: medium
  Step 9: Protein inference and summarization
    tool: custom R pipeline
    algorithm: top-N selection (top-3 peptides, top-3–6 fragment ions)
    parameters: peptide_selection=top 3 most intense peptides per protein, fragment_selection=['top 3 (S. pyogenes)', 'top 5 (plasma)', 'top 6 (multisite)'], singleton_removal=True, completeness_threshold=0.4, transition_minimum=4950, run_filtering=exclude runs < μₙ − 1.96·σₙ or low TIC
    input: normalized peptide matrix
    output: protein-level quantitative matrix
    confidence: medium
  Step 10: Differential proteomics (cross-sectional)
    tool: nlme (R package)
    algorithm: linear mixed-effects model (LME)
    parameters: model_formula=intensity ~ IRIS + peptideID + (1|Batch) + (1|AcqOrder) + (1|ID), method=ML for p-value, REML for effect size, p_adjustment=Benjamini-Hochberg, significance_threshold_p=0.05, effect_size_threshold=log2(1.25)
    input: protein-level matrix + metadata (IRIS status, Batch, AcqOrder, ID)
    output: list of differentially abundant proteins (IR/RVI)
    confidence: high
  Step 11: Differential proteomics (longitudinal RVI)
    tool: nlme (R package)
    algorithm: linear mixed-effects model with time-event factor
    parameters: model_formula=intensity ~ event + peptideID + (1|Batch) + (1|AcqOrder) + (1|ID), event_levels=5, p_adjustment=Benjamini-Hochberg, significance_threshold_p=0.05
    input: protein-level matrix + longitudinal metadata (event, Batch, AcqOrder, ID)
    output: list of longitudinally changing proteins
    confidence: high
  Step 12: Functional enrichment analysis
    tool: IMPaLA
    algorithm: pathway over-representation analysis
    parameters: q_value_threshold=1.0, clustering_method=fuzzy c-means, clusters=4, minimum_cluster_score=0.6, standardization=z-score per peptide
    input: clustered protein lists from longitudinal analysis
    output: enriched biological pathways
    confidence: medium
