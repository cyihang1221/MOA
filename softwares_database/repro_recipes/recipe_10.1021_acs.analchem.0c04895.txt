workflow_steps: 5

    tool: MZmine
    algorithm: built-in converter
    input: vendor raw files (e.g., .raw, .d)
    output: .mzML or .mzXML
    confidence: medium
  Step 2: Feature detection with diagnostic fragmentation filtering
    tool: MZmine
    algorithm: DFBuilder
    parameters: fragmentation_patterns=user-defined list, mass_tolerance_ppm=5, rt_tolerance_sec=10
    input: converted .mzML/.mzXML files
    output: feature table (precursor ions matching diagnostic fragments)
    confidence: high
  Step 3: Feature alignment and gap filling
    tool: MZmine
    algorithm: join aligner
    parameters: mass_tolerance_ppm=5, rt_tolerance_sec=10, weighting=intensity
    input: feature lists from multiple samples
    output: aligned feature table with gap-filled intensities
    confidence: medium
  Step 4: Feature annotation (MS/MS spectral matching)
    tool: MZmine
    algorithm: library search (e.g., MassBank, GNPS)
    parameters: mass_tolerance_ppm=5, min_match_peaks=3, min_relative_intensity_percent=1
    input: aligned feature table + MS/MS spectra
    output: annotated adduct candidates with spectral matches
    confidence: low
  Step 5: Curation and validation of putative DNA adducts
    tool: manual curation + authentic standards
    algorithm: retention time and MS/MS match to reference standards
    parameters: rt_shift_threshold_sec=5, spectral_similarity_score_threshold=0.7
    input: annotated candidates + standard spectra/RTs
    output: curated list of high-confidence DNA adducts
    confidence: high
