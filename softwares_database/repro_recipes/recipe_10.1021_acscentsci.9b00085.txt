paper_title: Rapid Prediction of Electron-Ionization Mass Spectrometry Using Neural Networks
source_paper: Wei JN et al. (2019), ACS Central Science, DOI: 10.1021/acscentsci.9b00085, PMID: 31041390
overall_reproducibility_score: 87/100

workflow_steps: 6
  Step 1: Molecular fingerprint generation
    tool: RDKit
    algorithm: Extended Circular Fingerprints (ECFP)
    parameters: fingerprint_length=4096, radius=2, type=additive
    input: SMILES or molecular structure
    output: additive ECFP vector
    confidence: high
  Step 2: Forward spectral prediction
    tool: TensorFlow
    algorithm: Multilayer Perceptron (MLP) regression
    parameters: batch_size=100, training_steps=100000, optimizer=Adam, loss_function=weighted_mean_squared_error, weighting_scheme=m/z-weighted as in Eq. 1
    input: additive ECFP vector
    output: predicted intensity vector (discrete m/z bins, 1–500 Da)
    confidence: high
  Step 3: Reverse spectral prediction
    tool: TensorFlow
    algorithm: Reverse-indexed MLP (mass-residual parametrization)
    parameters: shift_parameter=small shift for isotopes, indexing_scheme=i → m_z - m_fragment, shared_hidden_layers=True, separate_final_layer=True
    input: additive ECFP vector + molecular mass
    output: reverse-predicted intensity vector (indexed by neutral loss)
    confidence: medium
  Step 4: Bidirectional spectral fusion
    tool: custom TensorFlow implementation
    algorithm: Coordinate-wise gated combination
    parameters: gate_activation=sigmoid, gate_input=ECFP features, fusion_rule=y_i = g_i * y_i^forward + (1 - g_i) * y_i^reverse
    input: forward prediction + reverse prediction + ECFP features
    output: bidirectional predicted spectrum
    confidence: medium
  Step 5: Postprocessing and physical constraints
    tool: custom Python/TensorFlow
    algorithm: Mass cutoff and intensity zeroing
    parameters: max_mz=molecular_mass, zero_out_above_mz=True
    input: bidirectional predicted spectrum
    output: physically constrained predicted spectrum
    confidence: high
  Step 6: Library augmentation and matching evaluation
    tool: custom evaluation script
    algorithm: Weighted cosine similarity (Eq. 1)
    parameters: similarity_metric=weighted_cosine, weighting=m/z, mass_filter_tolerance=few Daltons, recall_at_k=10
    input: query spectra (NIST 2017 replicates) + augmented library (NIST main + NEIMS predictions)
    output: ranked matches, recall@10 score
    confidence: high
