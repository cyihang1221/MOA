paper_title: Toward Automated Preprocessing of Untargeted LC-MS-Based Metabolomics Feature Lists from Human Biofluids.
source_paper: Hughes A et al. (2025), Analytical chemistry, DOI: 10.1021/acs.analchem.4c03124, PMID: 39757901
overall_reproducibility_score: 87/100

workflow_steps: 9
  Step 1: LC-MS data conversion to mzML
    tool: ProteoWizard
    algorithm: msconvert
    input: raw vendor files (.raw, .wiff, .d)
    output: centroided .mzML files
    confidence: medium
  Step 2: Feature detection (peak picking)
    tool: XCMS
    algorithm: centWave
    parameters: ppm=10, snthr=10, prefilter=[3, 100], noise=10000, fwht=10, mzdiff=-0.001, mtol=25
    input: centroided .mzML files
    output: feature table (m/z, RT, intensity)
    confidence: high
  Step 3: Retention time alignment and correction
    tool: XCMS
    algorithm: loess
    parameters: span=0.3, family=symmetric
    input: feature table
    output: aligned feature table
    confidence: high
  Step 4: Peak grouping and deconvolution
    tool: XCMS
    algorithm: density
    parameters: bw=15, minfrac=0.5, mzwid=0.015, max=50
    input: aligned feature table
    output: grouped feature table
    confidence: high
  Step 5: Automated parameter optimization
    tool: IPO
    algorithm: iterative parameter optimization
    parameters: niter=10, method=cv, cv_threshold=0.3, qc_filtering=True
    input: QC sample feature tables
    output: optimized XCMS parameter set
    confidence: high
  Step 6: Automated parameter optimization
    tool: AutoTuner
    algorithm: genetic algorithm
    parameters: population_size=50, max_generations=100, fitness_metric=feature_count_cv30
    input: QC sample feature tables
    output: optimized XCMS parameter set
    confidence: high
  Step 7: Feature filtering using QC-based PCA
    tool: R base + stats
    algorithm: PCA on QC samples + variable loadings thresholding
    parameters: n_components=2, loading_threshold=0.1, cv_threshold=0.3
    input: feature table with QC samples
    output: filtered feature table
    confidence: medium
  Step 8: Tentative metabolite identification
    tool: HMDB
    algorithm: exact mass matching + retention time index (if available)
    parameters: mass_tolerance_ppm=10, rt_tolerance_sec=30, database=HMDB v5.0
    input: filtered feature table (m/z, RT)
    output: annotated feature table (HMDB IDs, names, classes)
    confidence: medium
  Step 9: Feature utility assessment
    tool: R
    algorithm: intensity binning and CV calculation
    parameters: intensity_bins=[1000.0, 10000.0, 100000.0, 1000000.0], cv_threshold=0.3
    input: filtered feature table
    output: utility metrics (CV%, bin occupancy, annotation rate)
    confidence: high
