    tool: DeepSeMS
    algorithm: Tenfold cross-validated transformer ensemble
    parameters: cross_validation_folds=10, ensemble_aggregation=prediction aggregation across folds, chemical_validity_rate=0.8571, mean_structural_similarity_to_reference=0.85
    input: Augmented BGC–SMILES training dataset

    confidence: high

    tool: RDKit
    algorithm: Descriptor calculation
    parameters: computed_properties=['molecular_weight', 'heavy_atom_count', 'synthetic_accessibility']

    output: Annotated metabolite predictions with physicochemical and novelty metrics
    confidence: medium
  Step 5: Global ocean metagenome screening
    tool: DeepSeMS
    algorithm: Large-scale BGC-to-SM prediction
    parameters: input_bgc_source=bathypelagic MAGs from global ocean metagenomes, predicted_metabolites_count=60000
    input: Cryptic BGCs from ocean MAGs
    output: Predicted secondary metabolite structures and annotations
    confidence: high
