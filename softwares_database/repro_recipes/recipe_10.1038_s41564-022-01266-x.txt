paper_title: Standardized multi-omics of Earth's microbiomes reveals microbial and metabolite diversity.
source_paper: Shaffer JP et al. (2022), Nature microbiology, DOI: 10.1038/s41564-022-01266-x, PMID: 36443458
overall_reproducibility_score: 72/100

workflow_steps: 8
  Step 1: Data conversion from vendor format to mzML
    tool: ProteoWizard (v3.0.19)
    algorithm: MSConvert
    input: raw .raw files
    output: centroided .mzML files
    confidence: high
  Step 2: Feature detection and chromatogram building
    tool: MZmine 2
    algorithm: ADAP chromatogram builder
    parameters: algorithm=ADAP, peak_detection_method=local_minimum_search
    input: centroided .mzML files
    output: feature list (peaks)
    confidence: medium
  Step 3: Isotopologue and adduct grouping
    tool: MZmine 2
    algorithm: Ion-identity networking
    parameters: ion_identity_networking_enabled=True
    input: detected features
    output: grouped features (molecular entities)
    confidence: medium
  Step 4: Retention time alignment and peak alignment across samples
    tool: MZmine 2
    algorithm: Join aligner
    parameters: min_peak_height=10000, weight_for_rt=10.0, weight_for_mz=10.0, weight_for_intensity=0.0
    input: features per sample
    output: aligned feature table
    confidence: low
  Step 5: Gap filling and feature filtering
    tool: MZmine 2
    algorithm: Gap filler
    parameters: min_fraction=0.3, min_samples=3, require_ms2=True
    input: aligned feature table
    output: filtered feature table with gap-filled intensities
    confidence: high
  Step 6: Feature-based molecular networking and spectral annotation
    tool: GNPS
    algorithm: Feature-Based Molecular Networking (FBMN)
    parameters: cosine_score_threshold=0.7, min_matched_peaks=6, max_mass_diff=0.5, library_matching_score_threshold=0.5
    input: .MGF spectra + .CSV feature table
    output: molecular families, putative annotations, spectral matches
    confidence: medium
  Step 7: Metabolite superclass and pathway assignment
    tool: ClassyFire
    algorithm: Chemical classification
    parameters: max_classification_depth=5
    input: putatively annotated metabolites (SMILES/InChI)
    output: superclass, class, subclass annotations
    confidence: low
  Step 8: Statistical association and co-occurrence analysis
    tool: R (custom scripts)
    algorithm: Mantel test, machine learning (XGBoost), correlation networks
    parameters: mantel_method=pearson, mantel_permutations=999, xgboost_ntree=500, xgboost_maxdepth=6
    input: normalized feature table + taxonomic/functional tables
    output: association p-values, feature importance rankings, co-occurrence networks
    confidence: low
