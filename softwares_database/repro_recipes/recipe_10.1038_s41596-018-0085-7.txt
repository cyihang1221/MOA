paper_title: A comprehensive pipeline for translational top-down proteomics from a single blood draw.
source_paper: Toby TK et al. (2019), Nature protocols, DOI: 10.1038/s41596-018-0085-7, PMID: 30518910
overall_reproducibility_score: 58/100

workflow_steps: 7
  Step 1: Sample collection and PBMC preservation
    tool: Clinical biorepository protocols
    input: whole blood
    output: preserved PBMCs (cryopreserved or fresh)
    confidence: medium
  Step 2: Cell lysis
    tool: Standard lysis buffer
    parameters: buffer_composition=not specified, protease_inhibitors=used
    input: PBMCs
    output: intact protein lysate
    confidence: low
  Step 3: Proteoform pre-fractionation
    tool: GELFrEE (gel-eluted liquid fractionation entrapment electrophoresis)
    parameters: fraction_number=12, molecular_weight_range=20–250 kDa
    input: intact protein lysate
    output: fractionated intact proteoforms
    confidence: high
  Step 4: LC-MS/MS acquisition
    tool: High-resolution Fourier-transform mass spectrometer
    algorithm: data-dependent top-two method
    parameters: duty_cycle=shown in Fig. 5, lc_valve_setup=Fig. 4, ms_resolution=high-resolution FT-MS, fragmentation_mode=HCD or ETD
    input: GELFrEE fractions
    output: .raw files (Thermo)
    confidence: medium
  Step 5: Targeted data analysis
    tool: ProSight Lite
    algorithm: mass defect filtering, fragment ion matching
    parameters: mass_tolerance_ppm=10, search_mode=targeted, database=UniProtKB
    input: .raw files
    output: targeted proteoform identifications
    confidence: high
  Step 6: Discovery-mode data analysis
    tool: TDPortal
    algorithm: deconvolution, database search, spectral alignment
    parameters: mass_tolerance_ppm=10, search_mode=discovery, database=UniProtKB, proteoform_coverage_threshold=not specified
    input: .raw files
    output: high-throughput proteoform identifications
    confidence: medium
  Step 7: Data visualization and QC
    tool: Top Down Viewer
    input: search results (.xml/.csv) and .raw files
    output: interactive spectra, chromatograms, QC metrics
    confidence: high
