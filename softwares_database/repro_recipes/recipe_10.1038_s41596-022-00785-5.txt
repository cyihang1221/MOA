workflow_steps: 6
  Step 1: Proteolytic digestion
    tool: Trypsin
    parameters: digestion_time_min=120, temperature_celsius=37, pH=8.0
    input: intact biotherapeutic protein sample
    output: tryptic peptide mixture
    confidence: medium
  Step 2: Reversed-phase chromatographic separation
    tool: UHPLC
    parameters: column_type=C18, column_dimensions=['150 × 2.1 mm', '250 × 2.1 mm'], particle_size_um=[3.0, 2.2], gradient_profile=not fully specified, flow_rate_ml_min=not specified, temperature_celsius=not specified
    input: tryptic peptide mixture
    output: chromatographically separated peptides
    confidence: low
  Step 3: High-resolution LC-MS analysis (discovery phase)
    tool: High-resolution mass spectrometer
    parameters: mass_resolution=high_resolution, mass_tolerance_ppm=not specified, scan_mode=data-dependent acquisition (DDA), ionization_mode=electrospray ionization (ESI)
    input: chromatographically separated peptides
    output: raw .raw or .d files (vendor format)
    confidence: low
  Step 4: Peptide identification and library creation
    tool: Proteomics search engine (e.g., Byonic, Proteome Discoverer, or similar)
    algorithm: database search against biopharmaceutical primary sequence
    parameters: enzyme=trypsin, max_missed_cleavages=2, fixed_modifications=['carbamidomethylation of C'], variable_modifications=['oxidation of M', 'deamidation of N/Q'], mass_tolerance_ppm=not specified, FDR_threshold=0.01
    input: raw MS data + protein sequence database
    output: targeted peptide library (retention time + accurate mass)
    confidence: medium
  Step 5: Targeted monitoring phase with differential analysis
    tool: Targeted LC-MS software (e.g., Thermo Compound Discoverer, Skyline, or equivalent)
    algorithm: new peak detection via differential analysis
    parameters: rt_window_min=0.5, mass_tolerance_ppm=not specified, threshold_for_change=not quantified, baseline_correction=implied
    input: raw MS data + targeted peptide library
    output: quantified PQAs (presence/absence/change of modifications/impurities)
    confidence: low
  Step 6: System suitability testing (SST)
    tool: NISTmAb tryptic digest reference standard
    parameters: acceptance_criteria=['peak shape', 'retention time reproducibility', 'signal-to-noise ratio', 'peak area precision'], columns_tested=['Hypersil Gold', 'Acclaim C18']
    input: reference standard digest
    output: SST report (pass/fail per attribute)
    confidence: high
