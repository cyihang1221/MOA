#配置Git身份（一次生效）
git config --global user.email "xxx@gmail.com"
git config --global user.name "用户名"
#克隆仓库
git clone git@github.com:cyihang1221/massagent.git #仓库地址可以在网站上的 <>code 界面复制

#添加文件，这里的实例是README.md 文件
git add README.md 
#提交上一步添加的文件，其中-m 后面的字符串是提交文件的描述内容
git commit -m "first commit"
#将提交的所有文件推送到仓库中，这里的main是分支名称，我推送到了tools-and-mcp_server分支中，后续和main合并。origin是远程仓库的默认别名（通常指克隆来源的那个远程仓库）
git push origin main

#查看本地所有分支
git branch
#切换分支
git checkout 分支名
git checkout -b ...
git remote add origin git@github.com:cyihang1221/MOA.git

git pull <remote> <branch>
