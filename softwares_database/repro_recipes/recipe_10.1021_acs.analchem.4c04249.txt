workflow_steps: 5
  Step 1: Fragment m/z normalization
    tool: PAH-Finder
    algorithm: m/z range normalization
    parameters: mz_range_min=0.0, mz_range_max=100.0, reference_peak=molecular_ion
    input: centroided MS/MS spectra (HRMS)
    output: normalized fragment intensity profile
    confidence: high
  Step 2: Feature extraction
    tool: PAH-Finder
    algorithm: seven custom PAH fragmentation features
    parameters: features=['relative_fragment_intensity_at_0-20%', 'relative_fragment_intensity_at_20-40%', 'relative_fragment_intensity_at_40-60%', 'relative_fragment_intensity_at_60-80%', 'relative_fragment_intensity_at_80-100%', 'presence_of_doubly_charged_fragments', 'molecular_formula_consistency_score']
    input: normalized fragment intensity profile
    output: feature vector per spectrum
    confidence: high
  Step 3: PAH classification
    tool: scikit-learn
    algorithm: random_forest
    parameters: n_estimators=100, max_depth=None, random_state=42, class_weight=balanced
    input: feature vectors
    output: PAH/non-PAH prediction + confidence score
    confidence: medium
  Step 4: Molecular formula prediction and filtering
    tool: PAH-Finder
    algorithm: elemental composition assignment with mass accuracy constraint
    parameters: mass_accuracy_ppm=5.0, elements_allowed=['C', 'H', 'O', 'N', 'S', 'Cl', 'Br'], max_atoms={'C': 40, 'H': 60, 'O': 8, 'N': 4, 'S': 2, 'Cl': 2, 'Br': 2}
    input: precursor m/z and MS/MS metadata
    output: candidate molecular formulas with PAH backbone likelihood
    confidence: high
  Step 5: Heteroatom-doped PAH identification
    tool: PAH-Finder
    algorithm: backbone-aware formula validation + fragment pattern matching
    parameters: heteroatom_tolerance=True, backbone_similarity_threshold=0.7
    input: candidate molecular formulas + normalized MS/MS
    output: annotated heteroatom-doped PAHs
    confidence: medium
