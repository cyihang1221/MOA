workflow_steps: 14
  Step 1: LC-MS data acquisition (HILIC, positive mode)
    tool: Thermo Q Exactive
    parameters: ion_mode=positive, mz_range=[70, 800], column=Atlantis HILIC, lc_system=Shimadzu Nexera X2 U-HPLC
    input: frozen plasma samples
    output: raw .raw files
    confidence: high
  Step 2: LC-MS data acquisition (amide, negative MRM mode)
    tool: Agilent 6490 triple quadrupole
    parameters: ion_mode=negative, acquisition_mode=MRM, column=Waters XBridge Amide, lc_system=Agilent 1290 Infinity LC
    input: frozen plasma samples
    output: raw .d files
    confidence: high
  Step 3: Feature detection and integration (known & unknown)
    tool: Progenesis QI
    algorithm: peak picking with retention time alignment
    parameters: rt_tolerance_min=0.1, mass_tolerance_ppm=5, min_peak_intensity=None, peak_width_min_sec=None, peak_width_max_sec=None
    input: raw .raw and .d files
    output: feature table (m/z, RT, intensity)
    confidence: medium
  Step 4: Targeted metabolite quantification (MRM)
    tool: Agilent MassHunter QQQ Quantitative Analysis (v68)
    algorithm: MRM integration with internal standard calibration
    parameters: integration_method=manual peak review + automated integration, internal_standards=['isotope-labeled standards']
    input: raw .d files
    output: quantified metabolite concentrations
    confidence: high
  Step 5: Quality control and filtering
    tool: in-house scripts / TraceFinder 3.3 (v3.3)
    algorithm: CV-based and missingness filtering
    parameters: cv_threshold_percent=30, missingness_threshold_percent=30, pooled_sample_interval_hilic=20, pooled_sample_interval_amide=10, scaling_method=linear scaling to nearest pooled sample
    input: feature table
    output: filtered feature table
    confidence: high
  Step 6: Retention time correction and batch normalization
    tool: Progenesis QI
    algorithm: alignment using pooled QC samples
    parameters: alignment_reference=pooled plasma, rt_shift_correction=linear interpolation
    input: filtered feature table
    output: aligned and normalized feature table
    confidence: medium
  Step 7: MS/MS data acquisition (DDA + inclusion list)
    tool: Thermo ID-X Orbitrap
    algorithm: AcquireX DDA + targeted inclusion list
    parameters: collision_energies_eV=[10, 25, 50], stepped_collision_energies_eV=[10, 20, 30, 40, 50], mass_tolerance_ppm_msms=5, rt_window_min=0.1, inclusion_list_partitions=24, sample_concentration_factor=10
    input: concentrated pooled plasma extracts
    output: raw .raw MS/MS files
    confidence: high
  Step 8: MS/MS data processing and spectral extraction
    tool: MSConvert + MSnbase (vv. 3.12)
    algorithm: EIC extraction + precursor matching + spectral aggregation
    parameters: mz_tolerance_ppm_msms=5, rt_tolerance_min=0.1, noise_fragment_removal_mz=173.46, noise_tolerance_ppm=30, aggregation_method=average intensity
    input: raw .raw MS/MS files
    output: parsed MS/MS spectra (.mgf, .ms)
    confidence: high
  Step 9: Metabolite annotation (MS1 + MS/MS)
    tool: GNPS + in-house bioinformatic pipeline
    algorithm: molecular networking + spectral library matching + structure prediction
    parameters: network_similarity_score_threshold=0.7, library_match_score_threshold=0.7, mass_tolerance_ppm_library=5, rt_tolerance_min_library=0.1
    input: parsed MS/MS spectra + feature table
    output: annotated metabolites (level 1–3)
    confidence: medium
  Step 10: Statistical preprocessing for GWAS
    tool: R (custom scripts)
    algorithm: log-transformation + z-scoring + residualization + inverse normal transformation
    parameters: covariates=['age', 'sex', 'batch', 'ancestry PCs 1–10'], residualization_method=linear regression, normalization_method=inverse normal transform
    input: aligned feature table
    output: normalized metabolite residuals
    confidence: high
  Step 11: Genome-wide association testing
    tool: GCTA (fastGWA)
    algorithm: linear mixed-effects model
    parameters: model=LMM with GRM, covariates=['age', 'sex', 'ancestry PCs 1–10'], significance_threshold=8e-11, mac_filter=5
    input: normalized metabolite residuals + genotype dosage matrix
    output: GWAS summary statistics (SNP, beta, se, p)
    confidence: high
  Step 12: mQTL fine-mapping and sentinel variant identification
    tool: SNPClip + custom R scripts
    algorithm: LD-based clumping (r² ≥ 0.8) + 1 Mb window merging
    parameters: ld_threshold_r2=0.8, window_size_kb=1000, clumping_method=SNPClip
    input: GWAS summary statistics
    output: sentinel SNPs and mQTL regions
    confidence: medium
  Step 13: Cross-cohort meta-analysis
    tool: R 'metagen'
    algorithm: inverse-variance weighted fixed-effects meta-analysis
    parameters: validation_threshold_p=0.05, effect_direction_consistency_required=True
    input: GWAS summary statistics from MESA and HERITAGE
    output: meta-analyzed association statistics
    confidence: high
  Step 14: Functional annotation and novelty assessment
    tool: PhenoScanner + GWAS Catalog + gnomAD + ClinVar
    algorithm: database lookup + regional overlap query
    parameters: phenscanner_build=38, phenscanner_p_threshold=5e-08, query_region_kb=1000, catalog_sources=['mQTL', 'GWAS Catalog']
    input: sentinel SNPs
    output: annotated mQTLs with novelty flags
    confidence: high
