workflow_steps: 8
  Step 1: Raw data conversion to mzML

    output: .mzML files
    confidence: high
  Step 2: Molecular networking
    tool: GNPS (vData Analysis workflow version release 30)
    algorithm: Cosine similarity network clustering
    parameters: precursor_mass_tolerance_da=0.02, fragment_mass_tolerance_da=0.02, min_matched_peaks=6, min_cluster_size=2, cosine_score_threshold=0.8, library_match_score_threshold=0.7, min_library_matched_peaks=6
    input: .mzML files
    output: MS/MS molecular network (graph)
    confidence: high
  Step 3: Network visualization
    tool: Cytoscape (v3.8.0)
    algorithm: Force-directed layout
    input: GNPS network output (edges/nodes tables)
    output: Interactive network graph
    confidence: medium
  Step 4: HR-UPLC-ESI-MS/MS acquisition
    tool: Dionex Ultimate3000 + Q-Exactive Plus
    algorithm: Full-scan MS and data-dependent MS2
    parameters: column=Luna Omega C18 (100 × 2.1 mm, 1.6 μm), gradient=[{'time_min': 0, 'percent_B': 5}, {'time_min': 1, 'percent_B': 5}, {'time_min': 7, 'percent_B': 97}, {'time_min': 9, 'percent_B': 97}, {'time_min': 10, 'percent_B': 5}, {'time_min': 13, 'percent_B': 5}], flow_rate_ml_min=0.3, injection_volume_ul=5, eluent_A=H2O + 0.1% formic acid, eluent_B=acetonitrile + 0.1% formic acid, washing_blanks=every ≤5 samples
    input: methanolic extracts
    output: raw .RAW files
    confidence: high
  Step 5: MS/MS spectral prediction & annotation
    tool: CFM-ID 4.0 (v4.0)
    algorithm: Fragmentation tree prediction
    parameters: mass_tolerance_ppm=None, ionization_mode=positive
    input: m/z and retention time of features
    output: putative structural annotations
    confidence: medium
  Step 6: MS/MS spectral prediction & annotation (alternative)
    tool: MassFrontier 8.0 (v8.0)
    algorithm: Rule-based fragmentation
    input: m/z and retention time of features
    output: putative structural annotations
    confidence: low
  Step 7: Metabolite quantification (semi-quantitative)
    tool: Thermo Xcalibur or custom script
    algorithm: Extracted ion chromatogram (EIC) peak area integration
    parameters: normalization=peak_area_per_mg_wet_weight_of_agar, technical_replicates=1
    input: HR-UPLC-MS raw data
    output: relative abundance table (metabolite × sample)
    confidence: medium
  Step 8: MALDI-IMS sample preparation and acquisition
    tool: Bruker UltrafleXtreme MALDI TOF/TOF
    algorithm: Imaging MS with raster scanning
    parameters: matrix=1:1 DHB:CHCA (20 mg/mL in ACN:MeOH:H2O 70:25:5), spraying_cycles=60, drying_time_per_cycle_s=30, incubation_time_per_cycle_s=10, spray_time_per_cycle_s=1, rotation=180° after 30 cycles
    input: ITO-coated microbial cultures on agar
    output: 2D ion intensity maps (.imzML)
    confidence: high
