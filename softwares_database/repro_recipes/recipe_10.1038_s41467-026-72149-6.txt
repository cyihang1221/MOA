workflow_steps: 5
  Step 1: Spectral vector encoding
    tool: Word2Vec
    algorithm: textualization + word embedding + pooling
    parameters: vector_dimension=512
    input: raw MS2 spectrum (mz/intensity pairs)
    output: 512-dimensional spectral vector
    confidence: medium
  Step 2: Template search via spectral similarity
    tool: custom spectral similarity search
    algorithm: cosine similarity on spectral vectors
    parameters: top_k_templates=5
    input: 512-dimensional spectral vector
    output: list of top-k structurally similar metabolite templates (SMILES or InChI)
    confidence: low
  Step 3: Template encoding into feature vectors
    tool: custom encoder
    algorithm: fingerprint + formula + formula difference encoding
    parameters: fingerprint_bits=3878, formula_bits=10, formula_diff_bits=10, total_vector_dim=3898
    input: SMILES/InChI of template metabolites + query MS2 spectrum
    output: 3898-dimensional structure-informed template vector
    confidence: low
  Step 4: Structure-to-structure generation
    tool: MetGenX (encoder-decoder neural network)
    algorithm: structure-informed encoder-decoder with template guidance
    parameters: generation_mode=['database-restricted', 'database-free'], reranking_model=ML-based structural similarity scorer
    input: 3898-dimensional template vector
    output: generated candidate SMILES structures
    confidence: medium
  Step 5: Candidate re-ranking
    tool: custom machine learning model
    algorithm: structural similarity scoring against retrieved templates
    input: generated SMILES candidates + template SMILES
    output: ranked list of candidate structures (top-1, top-5, etc.)
    confidence: low
