workflow_steps: 4
  Step 1: Gene ranking
    tool: GSEA-P

    parameters: ranking_metric=signal_to_noise, phenotype_labels=two-class
    input: gene expression matrix (e.g., microarray intensities) with phenotype labels
    output: ranked gene list L
    confidence: high
  Step 2: Enrichment Score (ES) calculation
    tool: GSEA-P
    algorithm: weighted Kolmogorov–Smirnov-like running-sum statistic
    parameters: weighting_exponent=1.0, step_weight=correlation_with_phenotype
    input: ranked gene list L and predefined gene set S
    output: Enrichment Score (ES)
    confidence: high
  Step 3: Statistical significance estimation
    tool: GSEA-P
    algorithm: phenotype-based permutation test
    parameters: num_permutations=1000, preserve_gene_correlation=True
    input: ES and permuted phenotype labels
    output: nominal P value
    confidence: high
  Step 4: Multiple testing correction
    tool: GSEA-P
    algorithm: False Discovery Rate (FDR)
    parameters: fdr_threshold=0.25
    input: list of ES values across all gene sets
    output: normalized enrichment score (NES) and FDR q-value
    confidence: high
