paper_title: iDIA-QC: AI-empowered data-independent acquisition mass spectrometry-based quality control.
source_paper: Gao H et al. (2025), Nature communications, DOI: 10.1038/s41467-024-54871-1, PMID: 39837863
overall_reproducibility_score: 72/100

workflow_steps: 5
  Step 1: Raw file conversion to mzXML
    tool: msConvert
    algorithm: ProteoWizard conversion
    input: vendor-specific raw files (.d, .raw, .wiff)
    output: mzXML files
    confidence: high
  Step 2: DIA data analysis and quantification
    tool: DIA-NN (v1.7.15)
    algorithm: Library-based DIA quantification
    parameters: fdr_threshold=0.01, quantification_strategy=robust LC (high accuracy) mode, library_matching=spectral library matching
    input: mzXML files + spectral libraries
    output: peptide/protein quantification tables (mainoutput.* files)
    confidence: high
  Step 3: QC metric extraction
    tool: Custom Python/R scripts
    algorithm: Feature extraction from mzXML and DIA-NN outputs
    parameters: missing_value_imputation=minimum per feature, feature_filtering_threshold=0.7, ms1_interpolation_points=1000, linear_interpolation=True
    input: mzXML files, DIA-NN mainoutput.* files, MGF files (for timsTOF Pro)
    output: 15-dimensional QC metric table per raw file
    confidence: medium
  Step 4: Machine learning model training
    tool: XGBoost
    algorithm: Gradient boosting decision tree
    parameters: evaluation_metric=AUC, missing_value_handling=imputed with minimum per feature, feature_filtering=remove features with >70% NA
    input: 15-dimensional QC metric table
    output: binary LC/MS quality classifier (qualified/unqualified)
    confidence: high
  Step 5: Statistical analysis and visualization
    tool: R (v4.1.2)
    algorithm: Hierarchical clustering (centroid method), t-SNE, CV calculation, heatmap generation
    parameters: clustering_method=centroid, cv_calculation=std / mean
    input: QC metric table, peptide/protein quantification tables
    output: heatmaps, t-SNE plots, agreement statistics, ROC curves
    confidence: high
