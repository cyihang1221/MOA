paper_title: TagGraph reveals vast protein modification landscapes from large tandem mass spectrometry datasets.
source_paper: Devabhaktuni A et al. (2019), Nature biotechnology, DOI: 10.1038/s41587-019-0067-5, PMID: 30936560
overall_reproducibility_score: 87/100

workflow_steps: 5
  Step 1: De novo sequence-based database querying
    tool: TagGraph
    algorithm: FM-indexed substring matching
    parameters: min_substring_length=3
    input: MS/MS spectra (peak lists)
    output: candidate database peptide matches with maximal substring overlap
    confidence: medium
  Step 2: Tag identification from substring alignment
    tool: TagGraph
    algorithm: continuous amino acid substring extraction (>2 residues)
    parameters: min_tag_length=3
    input: de novo sequence and candidate database peptides
    output: sequence tags (nodes)
    confidence: medium
  Step 3: Candidate match enumeration via graph-based pathfinding
    tool: TagGraph
    algorithm: longest path in directed acyclic graph (DAG)
    input: sequence tags and modification edges
    output: candidate peptide-modification assignments
    confidence: low
  Step 4: Probabilistic spectrum scoring
    tool: TagGraph
    algorithm: hierarchical Bayes model with Expectation-Maximization (EM)
    parameters: fdr_threshold=0.01, em_convergence_tolerance=0.0001, max_em_iterations=100
    input: candidate matches and MS/MS spectra
    output: peptide-spectrum matches (PSMs) with EM scores and posterior probabilities
    confidence: high
  Step 5: Global modification frequency refinement
    tool: TagGraph
    algorithm: iterative abundance-based re-annotation
    parameters: min_global_frequency=0.001
    input: all PSMs across dataset
    output: refined PTM annotations and site assignments
    confidence: low
