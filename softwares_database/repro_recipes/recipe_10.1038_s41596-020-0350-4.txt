paper_title: Comprehensive structural glycomic characterization of the glycocalyxes of cells and tissues.

overall_reproducibility_score: 42/100

workflow_steps: 7
  Step 1: Cell/tissue membrane fraction enrichment
    tool: ultracentrifugation
    parameters: centrifugation_speed_rpm=100000, duration_h=1.5, temperature_c=4
    input: cell or tissue homogenate
    output: enriched membrane fraction
    confidence: medium
  Step 2: Release and derivatization of N-glycans and O-glycans
    tool: enzymatic release (PNGase F, O-glycanase) + permethylation
    parameters: enzyme_concentration_ug_ml=1, incubation_time_h=16, temperature_c=37, permethylation_protocol=sodium hydroxide/methyl iodide
    input: enriched membrane fraction
    output: permethylated N- and O-glycans
    confidence: medium
  Step 3: Extraction and preparation of glycolipids
    tool: chloroform/methanol/water extraction
    parameters: solvent_ratio_chcl3_meoh_h2o=2:1:0.8, phase_separation_centifugation=3000 rpm, 10 min
    input: enriched membrane fraction
    output: intact glycolipids (glycosphingolipids)
    confidence: medium
  Step 4: NanoLC-MS/MS analysis
    tool: nanoLC coupled to high-resolution mass spectrometer
    parameters: column=C18 reversed-phase nano-column, gradient_profile=linear gradient from 5% to 40% B over 60 min, mobile_phase_A=0.1% formic acid in water, mobile_phase_B=0.1% formic acid in acetonitrile, flow_rate_nL_min=300, mass_analyzer=Orbitrap, resolution_mz=60000, scan_range_mz=500–2000, ms1_acquisition_mode=full scan, ms2_acquisition_mode=HCD fragmentation with normalized collision energy 25–35%
    input: permethylated glycans and intact glycolipids
    output: raw .raw files
    confidence: high
  Step 5: Glycan and glycolipid identification and annotation
    tool: in-house glycomics database search + manual spectral interpretation
    algorithm: mass tolerance matching + fragment ion pattern matching
    parameters: mass_tolerance_ppm=5, fragment_tolerance_da=0.02, minimum_fragment_ions_matched=3
    input: raw .raw files
    output: annotated glycan/glycolipid list with compositions and structures
    confidence: low
  Step 6: Quantitation of glycoforms and glycoconjugates
    tool: peak area integration (Xcalibur or in-house script)
    algorithm: label-free quantitation
    parameters: integration_method=extracted ion chromatogram (XIC), peak_width_s=10
    input: raw .raw files
    output: quantitative glycan/glycolipid abundance table
    confidence: medium
  Step 7: Site-specific glycoproteomic analysis
    tool: trypsin digestion + LC-MS/MS + glycopeptide search engine
    algorithm: Byonic or pGlyco
    parameters: digestion_time_h=16, enzyme=trypsin, search_mass_tolerance_ppm=5, variable_modifications=['Oxidation (M)', 'Deamidation (NQ)'], fixed_modifications=['Carbamidomethyl (C)'], glycan_mass_shifts=user-defined glycan library
    input: separate enriched membrane fraction
    output: site-specific glycopeptide identifications with glycoform assignments
    confidence: low
