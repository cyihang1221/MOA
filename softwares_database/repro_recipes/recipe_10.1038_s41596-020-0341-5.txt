workflow_steps: 6
  Step 1: Untargeted UHPLC-HRMS profiling of pooled sample
    tool: UHPLC-HRMS
    algorithm: multi-energy MS acquisition (positive and negative modes)
    parameters: collision_energies=multiple, ionization_mode=['positive', 'negative']
    input: pooled plasma sample mixture
    output: raw HRMS data (.raw or .d format)
    confidence: medium
  Step 2: Feature detection and alignment
    tool: XCMS
    algorithm: centWave or matchedFilter (not specified, but standard for XCMS)
    parameters: peak_width=None, mz_tolerance_ppm=5, snthresh=10
    input: raw HRMS data
    output: feature table (m/z, RT, intensity)
    confidence: low
  Step 3: Adduct and isotope annotation
    tool: CAMERA
    algorithm: grouping by adducts, isotopes, and biotransformations
    parameters: ppm=5, nSlaves=4
    input: XCMS feature table
    output: annotated feature groups
    confidence: medium
  Step 4: MRM transition selection
    tool: MRM-Ion Pair Finder
    algorithm: precursor-to-product ion pair identification from HRMS data
    parameters: retention_time_window=±0.2 min, mass_error_tolerance_ppm=10
    input: CAMERA-annotated features + HRMS spectra
    output: candidate MRM transitions (Q1/Q3, CE)
    confidence: low
  Step 5: Retention time calibration using internal standards
    tool: custom/internal standard-based correction
    algorithm: linear or spline interpolation using IS retention times
    parameters: internal_standards_count=set used (not enumerated), rt_window_for_IS_matching=±0.1 min
    input: candidate MRM transitions + IS retention times
    output: calibrated RT-aligned MRM method
    confidence: medium
  Step 6: Pseudotargeted quantification via UHPLC-TQMS
    tool: UHPLC-TQMS
    algorithm: Multiple Reaction Monitoring (MRM)
    parameters: dwell_time_ms=50, interchannel_delay_ms=5, CE_optimized_per_transition=True
    input: calibrated MRM method + individual plasma samples
    output: quantitative peak area matrix (metabolite × sample)
    confidence: medium
