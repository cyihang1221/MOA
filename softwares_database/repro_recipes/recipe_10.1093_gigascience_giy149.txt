workflow_steps: 8
  Step 1: Data ingestion and format standardization
    tool: PhenoMeNal platform (vv1.0)
    input: raw vendor files (.raw, .wiff, .d, etc.) or mzML
    output: standardized mzML files
    confidence: medium
  Step 2: Feature detection and alignment
    tool: XCMS (v3.8.0)
    algorithm: centWave
    parameters: ppm=10, peakwidth=[10, 60], snthresh=10, prefilter=[5, 10000], noise=1000, bw=30, mzdiff=-0.001, max=5
    input: mzML files
    output: feature table (ion features x samples)
    confidence: high
  Step 3: Retention time correction and peak alignment
    tool: XCMS (v3.8.0)
    algorithm: obiwarp
    parameters: profiling=density, span=0.3, center=median
    input: XCMS feature list
    output: aligned feature table
    confidence: high
  Step 4: Gap filling and missing value imputation
    tool: xcms (v3.8.0)
    algorithm: fillPeaks
    parameters: method=loess, max=3
    input: aligned feature table
    output: filled feature table
    confidence: medium
  Step 5: Quality control and filtering
    tool: CAMERA (v2.34.0)
    algorithm: annotate
    parameters: polarity=positive, ppm=5, mzabs=0.0025, rettime=15
    input: filled feature table + metadata
    output: QC-filtered feature table with adduct annotation
    confidence: medium
  Step 6: Statistical analysis and multivariate modeling
    tool: MetaboAnalystR (v3.0)
    algorithm: PLS-DA, OPLS-DA, PCA, ANOVA, FDR correction
    parameters: ncomp=2, perm=200, fdr_method=BH, log_transform=True, autoscale=True, cv=7
    input: QC-filtered feature table + sample metadata
    output: statistical results, model metrics, biomarker lists
    confidence: high
  Step 7: Metabolite identification and annotation
    tool: SIRIUS + CSI:FingerID (v4.2.0)
    algorithm: fragmentation tree analysis + molecular fingerprint prediction
    parameters: ms_level=2, adducts=['[M+H]+', '[M+Na]+'], mass_accuracy_ppm=5
    input: MS/MS spectra (mzML) + feature table
    output: putatively annotated metabolites with confidence scores
    confidence: medium
  Step 8: Pathway and enrichment analysis
    tool: MetaboAnalystR (v3.0)
    algorithm: hypergeometric test + impact analysis
    parameters: database=HMDB, organism=Homo sapiens, top_pathways=20, impact_threshold=0.1
    input: annotated metabolite list + pathway database mapping
    output: enriched pathways with p-values and impact scores
    confidence: high
