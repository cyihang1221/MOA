workflow_steps: 4
  Step 1: Feature grouping by m/z
    tool: metabCombiner
    algorithm: m/z tolerance-based grouping
    parameters: mz_tolerance_ppm=5
    input: feature tables (peak lists) from two untargeted LC-HRMS data sets
    output: m/z-aligned feature groups (candidate pairs)
    confidence: high
  Step 2: Retention time mapping via spline fitting
    tool: metabCombiner
    algorithm: cubic spline interpolation
    parameters: min_anchor_pairs=10, rt_ordering_required=True
    input: selected high-confidence m/z-matched feature pairs with ordered retention times
    output: retention time transformation function (RT warp model)
    confidence: medium
  Step 3: Feature pair scoring and ranking
    tool: metabCombiner
    algorithm: weighted similarity scoring (m/z + mapped RT + relative abundance)
    parameters: weight_mz=0.4, weight_rt=0.4, weight_abundance=0.2
    input: candidate feature pairs + RT warp model + normalized abundances
    output: ranked list of best-matching feature pairs
    confidence: high
  Step 4: Concatenation of intersecting features
    tool: metabCombiner
    algorithm: intersection-based table merging
    parameters: keep_all_features=False, merge_strategy=intersect
    input: top-ranked matched feature pairs + original abundance matrices
    output: combined feature abundance table (n features × (samples_A + samples_B))
    confidence: high
