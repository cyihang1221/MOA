workflow_steps: 4
  Step 1: Sample preparation with MALDI matrix selection
    tool: Manual preparation
    parameters: protein_ISD_matrix=1,5-diaminonaphthalene, glycan_ISD_matrix=2,5-dihydroxybenzoic acid / 2-hydroxy-5-methoxybenzoic acid (super-DHB), salt_condition_for_protein_analysis=absence of salts, salt_condition_for_glycan_analysis=presence of salt (for sodiated glycan detection)
    input: purified glycoconjugate vaccine candidates (E. coli O2/O6A/O25B conjugated to Pseudomonas exotoxin A)
    output: MALDI target spots ready for ISD FT-ICR MS analysis
    confidence: high
  Step 2: MALDI-in-source decay (ISD) FT-ICR mass spectrometry
    tool: FT-ICR MS
    parameters: instrument_type=Fourier transform ion cyclotron resonance mass spectrometer, ionization_mode=MALDI, decay_type=in-source decay (ISD), resolution_requirement=ultrahigh-resolution, mass_accuracy_requirement=sub-ppm
    input: MALDI-prepared samples
    output: raw ISD mass spectra (protein fragment ions and glycan fragment ions)
    confidence: medium
  Step 3: Selective detection and interpretation of ISD fragments
    tool: Manual spectral interpretation / custom annotation
    parameters: fragment_ion_types_interpreted=['protein N-terminal sequence ions', 'glycan repeat unit signature ions'], adducts_monitored=['sodiated glycan fragments']
    input: raw ISD FT-ICR MS spectra
    output: annotated protein terminal sequences and O-antigen repeat unit structures
    confidence: low
  Step 4: MS/MS analysis of O-antigen ISD fragments
    tool: Tandem MS (MS/MS) on FT-ICR platform
    parameters: target_ions=O-antigen ISD fragment ions, purpose=detection of specific repeat unit signatures
    input: selected ISD glycan fragment ions
    output: MS/MS spectra supporting repeat unit composition and linkage inference
    confidence: medium
