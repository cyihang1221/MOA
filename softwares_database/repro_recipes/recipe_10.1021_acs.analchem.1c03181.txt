paper_title: UniDecCD: Deconvolution of Charge Detection-Mass Spectrometry Data.

overall_reproducibility_score: 82/100

workflow_steps: 4
  Step 1: CD-MS data acquisition
    tool: custom charge detection mass spectrometer
    input: raw ion signal time traces (charge and m/z measurements per ion event)
    output: raw CD-MS event list (charge, m/z, intensity)
    confidence: low
  Step 2: Charge-state deconvolution
    tool: UniDecCD (v1.0)
    algorithm: iterative maximum likelihood deconvolution with charge uncertainty modeling
    parameters: charge_uncertainty_sigma=1.5, mass_resolution_target_da=100, max_charge_range=200, min_mass_da=10000, max_mass_da=50000000, bin_width_da=50
    input: raw CD-MS event list
    output: deconvolved mass spectrum (mass, intensity)
    confidence: high
  Step 3: Mass peak detection and annotation
    tool: UniDecCD (v1.0)
    algorithm: local maxima detection with signal-to-noise thresholding
    parameters: snr_threshold=3, peak_width_da=200, min_peak_distance_da=500
    input: deconvolved mass spectrum
    output: annotated mass peak table (mass_da, intensity, fwhm_da, snr)
    confidence: medium
  Step 4: Validation and resolution assessment
    tool: custom Python scripts
    algorithm: full width at half maximum (FWHM) calculation and mass accuracy evaluation
    parameters: reference_masses_da=['known_protein_masses', 'capsid_masses', 'nanodisc_composition_models']
    input: annotated mass peak table + reference masses
    output: resolution report (FWHM_da, mass_error_ppm, peak count)
    confidence: low
